   556→def esc(s: str) -> str:
   557→    """Escape HTML special chars in code samples."""
   558→    return s.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")
   559→
   560→
   561→def mini_term(code: str, title: str = "bash") -> str:
   562→    """Build a small terminal window.
   563→
   564→    `code` is the raw shell text. We escape it and add simple syntax
   565→    highlight spans for comments / prompts / output arrows.
   566→    """
   567→    e = esc(code)
   568→    lines = e.split("\n")
   569→    out = []
   570→    for ln in lines:
   571→        stripped = ln.lstrip()
   572→        if stripped.startswith("#"):
   573→            out.append(f'<span class="comment">{ln}</span>')
   574→        elif stripped.startswith("$ "):
   575→            inner = ln[2:]
   576→            out.append(f'<span class="prompt">$</span> <span class="cmd">{inner}</span>')
   577→        elif stripped.startswith("→ "):
   578→            inner = ln[2:]
   579→            out.append(f'<span class="arrow">→</span> <span class="out">{inner}</span>')
   580→        elif stripped.startswith("OUTPUT:"):
   581→            out.append(f'<span class="arrow">→</span> <span class="out">{ln}</span>')
   582→        else:
   583→            out.append(ln)
   584→    body = "\n".join(out)
   585→    return (
   586→        f'<div class="mini-term">'
   587→        f'<div class="bar"><i class="r"></i><i class="y"></i><i class="g"></i>'
   588→        f'<span class="t">{esc(title)}</span></div>'
   589→        f'<pre>{body}</pre></div>'
   590→    )
