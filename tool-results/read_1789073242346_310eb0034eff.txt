   323→
   324→def terminal(title: str, body_html: str) -> str:
   325→    """Build a terminal block with macOS-style title bar."""
   326→    return (
   327→        f'<div class="terminal">'

   337→
   338→

   341→    """Build a .card with header (number + icon + heading) and body."""

   343→        f'<div class="card head-{variant}">'
