   762→   762→    const connectorGrad = make("linearGradient", { id:"connectorGrad", x1:"0%", y1:"0%", x2:"100%", y2:"0%" });
   763→   763→    connectorGrad.appendChild(make("stop",{offset:"0%","stop-color":"#1a1a1a"}));
   764→   764→    connectorGrad.appendChild(make("stop",{offset:"40%","stop-color":"#050505"}));
   765→   765→    connectorGrad.appendChild(make("stop",{offset:"100%","stop-color":"#000000"}));
   766→   766→    defs.appendChild(connectorGrad);
   767→   767→
   768→   768→    /* Page-edge vertical shadow (fades right into paper) */
   769→   769→    const pageShadowGrad = make("linearGradient", { id:"pageEdgeShadow", x1:"0%", y1:"0%", x2:"100%", y2:"0%" });
   770→   770→    pageShadowGrad.appendChild(make("stop",{offset:"0%","stop-color":"#000000","stop-opacity":"0.16"}));
   771→   771→    pageShadowGrad.appendChild(make("stop",{offset:"40%","stop-color":"#000000","stop-opacity":"0.05"}));
   772→   772→    pageShadowGrad.appendChild(make("stop",{offset:"100%","stop-color":"#000000","stop-opacity":"0"}));
   773→   773→    defs.appendChild(pageShadowGrad);
   774→   774→
   775→   775→    svg.appendChild(defs);
   776→   776→
   777→   777→    /* ---- Page-edge vertical shadow (drawn once, full height) ---- */
   778→   778→    svg.appendChild(make("rect",{
   779→   779→      x:connectorX + connectorW - 2, y:0, width:12, height:height,
   780→   780→      fill:"url(#pageEdgeShadow)"
   781→   781→    }));
   782→   782→
   783→   783→    /* ---- Generate each U-shaped binding unit ---- */
   784→   784→    for (let i = 0; i < count; i++) {
   785→   785→      const cy = start + i * pitch;
   786→   786→
   787→   787→      /* The two horizontal wire strands — vertically centered around cy */
   788→   788→      const wire1Y = cy - wireGap/2;   // upper strand Y
   789→   789→      const wire2Y = cy + wireGap/2;   // lower strand Y
   790→   790→      const bendStartX = wireLeft + bendWidth;   // x where the bend begins
   791→   791→
   792→   792→      /* === LAYER 1: U-shaped wire — drawn as SEPARATE elements for reliable rendering ===
   793→   793→         1. Upper horizontal wire (line)
   794→   794→         2. Lower horizontal wire (line)
   795→   795→         3. Left rounded bend (Bézier path connecting the two wires)
   796→   796→         All solid black (#080808), same stroke width. */
   797→   797→
   798→   798→      /* Upper horizontal wire — thin rect for reliable rendering */
   799→   799→      svg.appendChild(make("rect",{
   800→   800→        x:bendStartX, y:wire1Y - wireStroke/2,
   801→   801→        width:wireRight - bendStartX, height:wireStroke,
   802→   802→        rx:wireStroke/2, fill:"#080808"
   803→   803→      }));
   804→   804→      /* Lower horizontal wire — thin rect for reliable rendering */
   805→   805→      svg.appendChild(make("rect",{
   806→   806→        x:bendStartX, y:wire2Y - wireStroke/2,
   807→   807→        width:wireRight - bendStartX, height:wireStroke,
   808→   808→        rx:wireStroke/2, fill:"#080808"
   809→   809→      }));
   810→   810→      /* Left rounded bend — Bézier connecting the two wire ends */
   811→   811→      svg.appendChild(make("path",{
   812→   812→        d:`M ${bendStartX} ${wire1Y} C ${wireLeft} ${wire1Y}, ${wireLeft} ${wire2Y}, ${bendStartX} ${wire2Y}`,
   813→   813→        fill:"none", stroke:"#080808", "stroke-width":wireStroke,
   814→   814→        "stroke-linecap":"round", "stroke-linejoin":"round"
   815→   815→      }));
   816→   816→
   817→   817→      /* === LAYER 2: Very subtle top-edge highlight (matte, not shiny) ===
   818→   818→         Thin lighter line along the upper edge of the upper strand only. */
   819→   819→      svg.appendChild(make("line",{
   820→   820→        x1:bendStartX + 1, y1:wire1Y - wireStroke/2 + 0.5,
   821→   821→        x2:wireRight - 2, y2:wire1Y - wireStroke/2 + 0.5,
   822→   822→        stroke:"#3a3a3a", "stroke-width":"0.5",
   823→   823→        "stroke-linecap":"round", opacity:"0.45"
   824→   824→      }));
   825→   825→
   826→   826→      /* === LAYER 3: Rounded black connector (FRONT — covers wire ends) ===
   827→   827→         Single shared cap that both wire ends enter.
   828→   828→         Small, rounded, NOT a perfect circle. */
   829→   829→      /* Connector shadow (subtle, behind, offset down-right) */
   830→   830→      svg.appendChild(make("rect",{
   831→   831→        x:connectorX, y:cy - connectorH/2, width:connectorW, height:connectorH,
   832→   832→        rx:connectorRx, fill:"#000", opacity:"0.35",
   833→   833→        transform:"translate(0.8 1.2)"
   834→   834→      }));
   835→   835→      /* Main connector body — solid black */
   836→   836→      svg.appendChild(make("rect",{
   837→   837→        x:connectorX, y:cy - connectorH/2, width:connectorW, height:connectorH,
   838→   838→        rx:connectorRx, fill:"#080808",
   839→   839→        stroke:"#000", "stroke-width":"0.4"
   840→   840→      }));
   841→   841→      /* Connector upper-left highlight (subtle 3D, not glossy) */
   842→   842→      svg.appendChild(make("rect",{
   843→   843→        x:connectorX + 1.5, y:cy - connectorH/2 + 1, width:connectorW - 3, height:2,
   844→   844→        rx:connectorRx - 1, fill:"#2a2a2a", opacity:"0.55"
   845→   845→      }));
   846→   846→      /* Connector bottom shadow */
   847→   847→      svg.appendChild(make("rect",{
   848→   848→        x:connectorX + 1.5, y:cy + connectorH/2 - 3, width:connectorW - 3, height:2,
   849→   849→        rx:connectorRx - 1, fill:"#000", opacity:"0.8"
   850→   850→      }));
   851→   851→    }
   852→   852→    binding.appendChild(svg);
   853→   853→
   854→   854→    /* Workaround for headless browser rendering bug:
   855→   855→       preserveAspectRatio="none" causes SVG content at x>28 to not render.
   856→   856→       Remove it AFTER appending, then force a re-render. */
   857→   857→    svg.removeAttribute("preserveAspectRatio");
   858→   858→    /* Force re-render */
   859→   859→    svg.style.display = "none";
   860→   860→    svg.offsetHeight;
   861→   861→    svg.style.display = "";
   862→   862→  }
   863→   863→
   864→   864→  /* Rebuild on resize so binding always matches page height */
   865→   865→  let timer;
   866→   866→  function rebuild() {
   867→   867→    clearTimeout(timer);
   868→   868→    timer = setTimeout(buildBinding, 50);
   869→   869→  }
   870→   870→  new ResizeObserver(rebuild).observe(document.querySelector(".page"));
   871→   871→  buildBinding();
   872→   872→})();
