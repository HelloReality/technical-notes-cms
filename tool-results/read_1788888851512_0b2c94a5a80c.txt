    87→// ─── Tree node type ───────────────────────────────────────────
    88→interface TreeNode {
    89→  category: Category;
    90→  topics: { category: Category; notes: Note[] }[];
    91→  notes: Note[];

    93→
    94→function buildTree(categories: Category[], notes: Note[]): TreeNode[] {
    95→  const top = categories.filter((c) => !c.parentId);
    96→  return top
    97→    .map((cat) => {
    98→      const topics = categories
    99→        .filter((c) => c.parentId === cat.id)
   100→        .map((sub) => ({
   101→          category: sub,
   102→          notes: notes
   103→            .filter((n) => n.categoryId === sub.id)
   104→            .sort(
   105→              (a, b) =>
   106→                new Date(a.createdAt).getTime() -
   107→                new Date(b.createdAt).getTime(),
   108→            ),
   109→        }))
   110→        .filter((t) => t.notes.length > 0);
   111→      const directNotes = notes
   112→        .filter((n) => n.categoryId === cat.id)
   113→        .sort(
   114→          (a, b) =>
   115→            new Date(a.createdAt).getTime() -
   116→            new Date(b.createdAt).getTime(),
   117→        );
   118→      return { category: cat, topics, notes: directNotes };
   119→    })
   120→    .filter((n) => n.topics.length > 0 || n.notes.length > 0);
   121→}

   414→  // ─── Note scaling (unified page object) ───────────────────────
   415→  // The ENTIRE notebook sheet — spiral rings + ring shadows + page

   417→  // margin — is treated as ONE renderable canvas. We measure the
   418→  // note's true bounding box (including absolutely-positioned rings
   419→  // and the box-shadow overflow), size the iframe to that exact box,
   420→  // then scale the whole object as a unit to fit the viewport width.
   421→  //
   422→  // The note's internal design is NEVER modified — we only neutralize
   423→  // the body's outer padding (which is decorative spacing for the raw
   424→  // standalone file) so the page centers cleanly in the iframe. No
   425→  // reflow, no redesign. Only the scale of the complete page object
   426→  // adjusts responsively per device.
   427→  const applyNoteScale = React.useCallback(() => {
   428→    const iframe = iframeRef.current;
   429→    const container = mainRef.current;
   430→    if (!iframe || !container) return;
   431→    try {
   432→      const doc = iframe.contentDocument;
   433→      if (!doc || !doc.body) return;
   434→
   435→      // Preserve the body's padding so the spiral binding rings
   436→      // (positioned at left:-30px on .page-wrapper) have enough
   437→      // space to be fully visible. The template CSS sets
   438→      // body{padding:24px 48px} — we must NOT override it.
   439→      // (Previously this neutralized padding which clipped the rings.)
   440→
   441→      const avail = container.clientWidth;
   442→      // Small intentional margin so the shadow/rings don't touch the
   443→      // viewport edges. Kept constant in screen px (not scaled).
   444→      const VIEWPORT_MARGIN = 16;
   445→      const targetW = Math.max(0, avail - VIEWPORT_MARGIN * 2);
   446→
   447→      // Find the page element (.page-wrapper / .page / .notebook).
   448→      const pageEl =
   449→        (doc.querySelector(".page-wrapper") as HTMLElement | null) ||
   450→        (doc.querySelector(".page") as HTMLElement | null) ||
   451→        (doc.querySelector(".notebook") as HTMLElement | null) ||
   452→        (doc.body.firstElementChild as HTMLElement | null) ||
   453→        doc.body;
   454→
   455→      // Read the page's CSS width directly (the note uses a fixed
   456→      // width:1080px on .page-wrapper). This is reliable regardless of
   457→      // the iframe's current laid-out width, avoiding chicken-and-egg
   458→      // measurement issues.
   459→      const computedW = doc.defaultView?.getComputedStyle(pageEl).width;
   460→      const parsedW = computedW ? parseFloat(computedW) : 0;
   461→      const pageWNum = isFinite(parsedW) && parsedW > 0 ? parsedW : 1080;
   462→
   463→      // The complete page object width = page + ring overflow (left) +
   464→      // shadow spread (both sides). The rings extend ~40px left of the
   465→      // page; the box-shadow spreads ~60px on all sides.
   466→      const RING_OVERFLOW = 40;
   467→      const SHADOW = 60;
   468→      const objectW = pageWNum + RING_OVERFLOW + SHADOW * 2;
   469→
   470→      // Body height includes the page; add shadow for the bottom.
   471→      const bodyH = Math.max(
   472→        doc.body.scrollHeight,
   473→        doc.body.offsetHeight,
   474→        doc.documentElement.scrollHeight,
   475→        doc.documentElement.offsetHeight,
   476→      );
   477→      const objectH = bodyH + SHADOW;
   478→
   479→      // Base fit-to-width scale (page object fills the viewport).
   480→      const fitScale = targetW < objectW ? targetW / objectW : 1;
   481→      // Apply the user's zoom multiplier. In "fit" mode zoom stays at 1
   482→      // (fitScale alone). In "manual" mode the user's zoom (0.25–4)
   483→      // scales relative to the fit width.
   484→      const zoomMul = zoomMode === "manual" ? zoom : 1;
   485→      const scale = fitScale * zoomMul;
   486→
   487→      // Size the iframe to the complete page object.
   488→      const NATIVE_W = objectW;
   489→      iframe.style.width = `${NATIVE_W}px`;
   490→      iframe.style.height = `${objectH}px`;
   491→      iframe.style.transform = `scale(${scale})`;
   492→      iframe.style.transformOrigin = "top left";
   493→
   494→      // Reserve the scaled footprint on the wrapper so the page can
   495→      // scroll vertically and centers horizontally. When zoomed in
   496→      // beyond the viewport width, the wrapper grows wider than the
   497→      // container and the scroll container handles horizontal scroll.
   498→      const wrapper = iframe.parentElement;
   499→      if (wrapper) {
   500→        wrapper.style.width = `${NATIVE_W * scale}px`;
   501→        wrapper.style.height = `${objectH * scale}px`;
   502→        wrapper.style.marginLeft = "auto";
   503→        wrapper.style.marginRight = "auto";
   504→      }
   505→    } catch {
   506→      /* cross-origin — ignore */
   507→    }
   508→  }, [zoom, zoomMode]);
   509→
   510→  // Re-scale when the viewport changes (resize, tree toggle, orientation).
   511→  React.useEffect(() => {
   512→    const handler = () => applyNoteScale();
   513→    window.addEventListener("resize", handler);
   514→    return () => window.removeEventListener("resize", handler);
   515→  }, [applyNoteScale]);
