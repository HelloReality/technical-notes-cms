   866→function conceptCard(c: Concept): string {
   867→  const tint = c.tint ?? "neutral";
   868→  const numHtml = c.num ? `<span class="num">${c.num}</span>` : "";
   869→  const iconHtml = c.icon ? `<span class="icn-mini bg-${tint}" style="color:var(--linux-green);">${c.icon}</span>` : "";
   870→  let bodyHtml = "";
   871→  if (c.body) bodyHtml += `<div class="body-text">${c.body}</div>`;
   872→  if (c.bullets && c.bullets.length) {
   873→    bodyHtml += `<ul>${c.bullets.map((b) => `<li>${b}</li>`).join("")}</ul>`;
   874→  }
   875→  if (c.cmds && c.cmds.length) {
   876→    for (const cmd of c.cmds) {
   877→      const term = cmd.cmd.replace(/\$/g, '<span class="pmt">$</span>');
   878→      const out = cmd.out ? `\n<span class="out">${esc(cmd.out)}</span>` : "";
   879→      const cmt = cmd.desc ? `\n<span class="cmt"># ${esc(cmd.desc)}</span>` : "";
   880→      bodyHtml += `<div class="mini-term">${term}${cmt}${out}</div>`;
   881→    }
   882→  }
   883→  return `<div class="concept-card">
   884→    <div class="ch">${numHtml}${iconHtml}<h3>${c.title}</h3></div>
   885→    ${bodyHtml}
   886→  </div>`;
   887→}

   889→function cmdCard(opts: {
   890→  cmd: string;
   891→  title?: string;
   892→  desc?: string;

   911→  }

   922→function termBlock(title: string, lines: string[]): string {
   923→  // Each line: escape, replace $ with prompt span, # with comment span
   924→  const body = lines
   925→    .map((ln) => {
   926→      // handle prompt
   927→      let out = esc(ln);
   928→      // highlight $ at start as prompt
   929→      out = out.replace(/^(\s*)\$/, '$1<span class="pmt">$</span>');
   930→      // highlight # at start as comment
   931→      out = out.replace(/^(\s*)#/, '$1<span class="cmt">#</span>');
   932→      return out;
   933→    })
   934→    .join("\n");
   935→  return `<div class="term-block">
   936→    <div class="term-head">
   937→      <span class="dot r"></span><span class="dot y"></span><span class="dot g"></span>
   938→      <span class="ttl">${esc(title)}</span>
   939→    </div>
   940→    <pre class="term-body">${body}</pre>
   941→  </div>`;
   942→}

   944→function dataTable(opts: {
   945→  caption?: string;
   946→  headers: string[];
   947→  rows: string[][];
   948→  monoCols?: number[];
   949→}): string {
   950→  const { caption, headers, rows } = opts;
   951→  const monoCols = new Set(opts.monoCols ?? []);
   952→  const head = headers
   953→    .map((h) => `<th>${esc(h)}</th>`)
   954→    .join("");
   955→  const body = rows
   956→    .map(
   957→      (r) =>
   958→        `<tr>${r
   959→          .map(
   960→            (cell, i) =>
   961→              `<td${monoCols.has(i) ? ' class="mono"' : ""}>${cell}</td>`,
   962→          )
   963→          .join("")}</tr>`,
   964→    )
   965→    .join("");
   966→  const table = `<table class="data-table">
   967→    <thead><tr>${head}</tr></thead>
   968→    <tbody>${body}</tbody>
   969→  </table>`;
   970→  if (!caption) return table;
   971→  return `<div class="table-wrap">
   972→    <div class="table-caption">${esc(caption)}</div>
   973→    ${table}
   974→  </div>`;
   975→}
