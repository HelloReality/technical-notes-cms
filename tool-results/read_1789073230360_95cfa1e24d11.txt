   561→   561→def mini_term(code: str, title: str = "bash") -> str:
   562→   562→    """Build a small terminal window.
   563→   563→
   564→   564→    `code` is the raw shell text. We escape it and add simple syntax
   565→   565→    highlight spans for comments / prompts / output arrows.
   566→   566→    """
   567→   567→    e = esc(code)
   568→   568→    lines = e.split("\n")
   569→   569→    out = []
   570→   570→    for ln in lines:
   571→   571→        stripped = ln.lstrip()

   575→   575→            inner = ln[2:]
   576→   576→            out.append(f'<span class="prompt">$</span> <span class="cmd">{inner}</span>')
   577→   577→        elif stripped.startswith("→ "):
   578→   578→            inner = ln[2:]
   579→   579→            out.append(f'<span class="arrow">→</span> <span class="out">{inner}</span>')
   580→   580→        elif stripped.startswith("OUTPUT:"):
   581→   581→            out.append(f'<span class="arrow">→</span> <span class="out">{ln}</span>')
   582→   582→        else:
   583→   583→            out.append(ln)
   584→   584→    body = "\n".join(out)
   585→   585→    return (
   586→   586→        f'<div class="mini-term">'
   587→   587→        f'<div class="bar"><i class="r"></i><i class="y"></i><i class="g"></i>'
