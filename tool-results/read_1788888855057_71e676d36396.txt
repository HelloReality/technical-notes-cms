    88→    88→interface TreeNode {

    90→    90→  topics: { category: Category; notes: Note[] }[];

    94→    94→function buildTree(categories: Category[], notes: Note[]): TreeNode[] {
    95→    95→  const top = categories.filter((c) => !c.parentId);
    96→    96→  return top
    97→    97→    .map((cat) => {
    98→    98→      const topics = categories
    99→    99→        .filter((c) => c.parentId === cat.id)
   100→   100→        .map((sub) => ({
   101→   101→          category: sub,
   102→   102→          notes: notes
   103→   103→            .filter((n) => n.categoryId === sub.id)
   104→   104→            .sort(
   105→   105→              (a, b) =>
   106→   106→                new Date(a.createdAt).getTime() -
   107→   107→                new Date(b.createdAt).getTime(),
   108→   108→            ),
   109→   109→        }))
   110→   110→        .filter((t) => t.notes.length > 0);
   111→   111→      const directNotes = notes
   112→   112→        .filter((n) => n.categoryId === cat.id)
   113→   113→        .sort(
   114→   114→          (a, b) =>
   115→   115→            new Date(a.createdAt).getTime() -
   116→   116→            new Date(b.createdAt).getTime(),
   117→   117→        );
   118→   118→      return { category: cat, topics, notes: directNotes };
   119→   119→    })
   120→   120→    .filter((n) => n.topics.length > 0 || n.notes.length > 0);
   121→   121→}
