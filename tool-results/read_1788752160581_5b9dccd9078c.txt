    88→    88→  const dataLoading = useAppStore((s) => s.dataLoading);
    89→    89→  const categories = useAppStore((s) => s.categories);
    90→    90→  const openUploadModal = useAppStore((s) => s.openUploadModal);
    91→    91→  const openNote = useAppStore((s) => s.openNote);
    92→    92→  const publishNote = useAppStore((s) => s.publishNote);
    93→    93→  const unpublishNote = useAppStore((s) => s.unpublishNote);
    94→    94→  const deleteNote = useAppStore((s) => s.deleteNote);
    95→    95→
    96→    96→  const [section, setSection] = React.useState<SidebarKey>("dashboard");

   100→   100→  );

   103→   103→

   106→   106→    const published = notes.filter((n) => n.status === "PUBLISHED").length;

   110→   110→  }, [notes]);

   124→   124→    });

   133→   133→        )
   134→   134→        .slice(0, 6),
   135→   135→    [notes],
   136→   136→  );
   137→   137→

   142→   142→        await unpublishNote(note.id);

   145→   145→        await publishNote(note.id);
   146→   146→        toast.success("Note published", { description: note.title });
   147→   147→      }
   148→   148→    } catch (err) {
   149→   149→      toast.error("Action failed", {
   150→   150→        description: err instanceof Error ? err.message : "Try again later.",
   151→   151→      });
   152→   152→    } finally {
   153→   153→      setBusyId(null);
   154→   154→    }
   155→   155→  };

   157→   157→  const handleDelete = async () => {

   163→   163→      toast.success("Note deleted", { description: target.title });
   164→   164→      setDeleteTarget(null);
   165→   165→    } catch (err) {
   166→   166→      toast.error("Delete failed", {
   167→   167→        description: err instanceof Error ? err.message : "Try again later.",
   168→   168→      });
   169→   169→    } finally {
   170→   170→      setBusyId(null);
   171→   171→    }

   182→   182→            (n) =>

   185→   185→              c.children?.some((ch) => n.categoryId === ch.id),

   187→   187→      }))

   198→   198→              value={section}
   199→   199→              onValueChange={(v) => setSection(v as SidebarKey)}
   200→   200→            >
   201→   201→              <SelectTrigger className="w-full">
   202→   202→                <SelectValue />
   203→   203→              </SelectTrigger>
   204→   204→              <SelectContent>
   205→   205→                {SIDEBAR_ITEMS.map((item) => (
   206→   206→                  <SelectItem key={item.key} value={item.key}>
   207→   207→                    {item.label}
   208→   208→                  </SelectItem>
   209→   209→                ))}

   211→   211→            </Select>

   214→   214→            aria-label="Admin sections"
   215→   215→            className="hidden flex-col gap-1 lg:flex"

   230→   230→          <div className="mt-4 hidden rounded-xl border border-border/60 bg-card p-4 lg:block">

   232→   232→              <ShieldCheck className="size-4 text-emerald-600 dark:text-emerald-400" />
   233→   233→              Admin Console
   234→   234→            </div>
   235→   235→            <p className="mt-1 text-xs text-muted-foreground">

   237→   237→            </p>

   239→   239→              className="mt-3 w-full gap-2 bg-emerald-600 text-white hover:bg-emerald-700 dark:bg-emerald-500 dark:hover:bg-emerald-600"

   244→   244→              Upload Note

   255→   255→              {SIDEBAR_ITEMS.find((i) => i.key === section)?.label}

   257→   257→            <p className="text-sm text-muted-foreground">
   258→   258→              Manage your technical knowledge base.
   259→   259→            </p>
   260→   260→          </div>
   261→   261→          <Button
   262→   262→            onClick={openUploadModal}
   263→   263→            className="gap-2 bg-emerald-600 text-white hover:bg-emerald-700 dark:bg-emerald-500 dark:hover:bg-emerald-600"

   266→   266→            Upload Note
   267→   267→          </Button>
   268→   268→        </div>
   269→   269→
   270→   270→        {section === "dashboard" && (
   271→   271→          <>
   272→   272→            <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 lg:grid-cols-4">
   273→   273→              <StatCard
   274→   274→                label="Total Notes"
   275→   275→                value={stats.total}
   276→   276→                icon={BookOpen}
   277→   277→                loading={dataLoading}
   278→   278→              />
   279→   279→              <StatCard
   280→   280→                label="Published"
   281→   281→                value={stats.published}
   282→   282→                icon={Eye}
   283→   283→                accent="emerald"
   284→   284→                loading={dataLoading}
   285→   285→              />
   286→   286→              <StatCard
   287→   287→                label="Drafts"
   288→   288→                value={stats.draft}
   289→   289→                icon={FileEdit}
   290→   290→                accent="amber"
   291→   291→                loading={dataLoading}
   292→   292→              />
   293→   293→              <StatCard
   294→   294→                label="Unpublished"
   295→   295→                value={stats.unpublished}
   296→   296→                icon={ShieldCheck}
   297→   297→                loading={dataLoading}
   298→   298→              />
   299→   299→            </div>
   300→   300→
   301→   301→            <Card>
   302→   302→              <CardHeader>
   303→   303→                <CardTitle className="text-base">Top categories</CardTitle>
   304→   304→                <CardDescription>
   305→   305→                  Distribution of notes across top-level categories.
   306→   306→                </CardDescription>
   307→   307→              </CardHeader>
   308→   308→              <CardContent>
   309→   309→                {dataLoading ? (
   310→   310→                  <div className="flex flex-col gap-3">
   311→   311→                    {Array.from({ length: 4 }).map((_, i) => (
   312→   312→                      <Skeleton key={i} className="h-8 w-full" />
   313→   313→                    ))}
   314→   314→                  </div>
   315→   315→                ) : topCategories.length === 0 ? (
   316→   316→                  <p className="text-sm text-muted-foreground">
   317→   317→                    No categories yet.
   318→   318→                  </p>
   319→   319→                ) : (
   320→   320→                  <div className="flex flex-col gap-2">
   321→   321→                    {topCategories.map((c) => {
   322→   322→                      const max = topCategories[0]?.count ?? 1;
   323→   323→                      const pct = max ? (c.count / max) * 100 : 0;
   324→   324→                      return (
   325→   325→                        <div key={c.id} className="flex items-center gap-3">
   326→   326→                          <div className="w-32 shrink-0 text-sm">{c.name}</div>

   328→   328→                            <div
   329→   329→                              className="absolute inset-y-0 left-0 rounded-full bg-emerald-500"
   330→   330→                              style={{ width: `${pct}%` }}
   331→   331→                            />
   332→   332→                          </div>
   333→   333→                          <div className="w-8 shrink-0 text-right text-xs text-muted-foreground">
   334→   334→                            {c.count}
   335→   335→                          </div>
   336→   336→                        </div>
   337→   337→                      );
   338→   338→                    })}
   339→   339→                  </div>
   340→   340→                )}
   341→   341→              </CardContent>
   342→   342→            </Card>
   343→   343→
   344→   344→            <Card>
   345→   345→              <CardHeader>
   346→   346→                <CardTitle className="text-base">Recent notes</CardTitle>
   347→   347→                <CardDescription>
   348→   348→                  The most recently updated notes across all categories.
   349→   349→                </CardDescription>
   350→   350→              </CardHeader>
   351→   351→              <CardContent className="px-0 pb-0">
   352→   352→                <NotesTable
   353→   353→                  notes={recentNotes}
   354→   354→                  loading={dataLoading}
   355→   355→                  onOpen={openNote}
   356→   356→                  onTogglePublish={handleTogglePublish}
   357→   357→                  onDelete={setDeleteTarget}
   358→   358→                  busyId={busyId}
   359→   359→                />
   360→   360→              </CardContent>
   361→   361→            </Card>
   362→   362→          </>
   363→   363→        )}
   364→   364→
   365→   365→        {section === "notes" && (
   366→   366→          <Card>
   367→   367→            <CardHeader>
   368→   368→              <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   369→   369→                <div>
   370→   370→                  <CardTitle className="text-base">All notes</CardTitle>
   371→   371→                  <CardDescription>
   372→   372→                    Filter, publish, unpublish, or delete notes.
   373→   373→                  </CardDescription>
   374→   374→                </div>
   375→   375→                <div className="flex flex-wrap items-center gap-2">
   376→   376→                  <div className="relative">
   377→   377→                    <Search className="pointer-events-none absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
   378→   378→                    <Input
   379→   379→                      value={search}
   380→   380→                      onChange={(e) => setSearch(e.target.value)}
   381→   381→                      placeholder="Filter…"
   382→   382→                      className="h-9 w-full pl-8 sm:w-56"
   383→   383→                    />
   384→   384→                  </div>
   385→   385→                  <Select
   386→   386→                    value={statusFilter}
   387→   387→                    onValueChange={(v) =>
   388→   388→                      setStatusFilter(v as NoteStatus | "ALL")
   389→   389→                    }
   390→   390→                  >
   391→   391→                    <SelectTrigger className="h-9 w-[140px]">
   392→   392→                      <SelectValue />
   393→   393→                    </SelectTrigger>
   394→   394→                    <SelectContent>
   395→   395→                      <SelectItem value="ALL">All statuses</SelectItem>
   396→   396→                      <SelectItem value="PUBLISHED">Published</SelectItem>
   397→   397→                      <SelectItem value="DRAFT">Draft</SelectItem>
   398→   398→                      <SelectItem value="UNPUBLISHED">Unpublished</SelectItem>
   399→   399→                    </SelectContent>
   400→   400→                  </Select>
   401→   401→                </div>
   402→   402→              </div>
   403→   403→            </CardHeader>
   404→   404→            <CardContent className="px-0 pb-0">
   405→   405→              <NotesTable
   406→   406→                notes={filteredNotes}
   407→   407→                loading={dataLoading}
   408→   408→                onOpen={openNote}
   409→   409→                onTogglePublish={handleTogglePublish}
   410→   410→                onDelete={setDeleteTarget}
   411→   411→                busyId={busyId}
   412→   412→              />
   413→   413→            </CardContent>
   414→   414→          </Card>
   415→   415→        )}
   416→   416→
   417→   417→        {section === "categories" && (
   418→   418→          <Card>
   419→   419→            <CardHeader>
   420→   420→              <CardTitle className="text-base">Categories</CardTitle>
   421→   421→              <CardDescription>
   422→   422→                Top-level categories and their subcategories.
   423→   423→              </CardDescription>
   424→   424→            </CardHeader>
   425→   425→            <CardContent>
   426→   426→              {categories.filter((c) => !c.parentId).length === 0 ? (
   427→   427→                <p className="text-sm text-muted-foreground">
   428→   428→                  No categories yet.
   429→   429→                </p>
   430→   430→              ) : (
   431→   431→                <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   432→   432→                  {categories
   433→   433→                    .filter((c) => !c.parentId)
   434→   434→                    .map((c) => {

   436→   436→                        (sub) => sub.parentId === c.id,
   437→   437→                      );
   438→   438→                      const count =
   439→   439→                        notes.filter(
   440→   440→                          (n) =>
   441→   441→                            n.categoryId === c.id ||
   442→   442→                            subs.some((s) => n.categoryId === s.id),
   443→   443→                        ).length ?? 0;
   444→   444→                      return (
   445→   445→                        <div
   446→   446→                          key={c.id}
   447→   447→                          className="rounded-xl border border-border/60 bg-card p-4"
   448→   448→                        >
   449→   449→                          <div className="flex items-center justify-between">
   450→   450→                            <div className="flex items-center gap-2">
   451→   451→                              <span className="flex size-8 items-center justify-center rounded-md bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300">

   455→   455→                                {c.name}
   456→   456→                              </span>

   459→   459→                          </div>

   465→   465→                                  variant="outline"

   469→   469→                                </Badge>

   471→   471→                            </div>
   472→   472→                          )}
   473→   473→                          {c.description && (

   475→   475→                              {c.description}
   476→   476→                            </p>

   478→   478→                        </div>
   479→   479→                      );
   480→   480→                    })}
   481→   481→                </div>

   486→   486→

   497→   497→            </CardContent>
   498→   498→          </Card>
   499→   499→        )}
   500→   500→
   501→   501→        {section === "settings" && <SettingsPanel />}
   502→   502→      </main>
   503→   503→

   737→   737→function TagsCloud({ notes }: { notes: Note[] }) {
   738→   738→  const tagMap = React.useMemo(() => {
   739→   739→    const m = new Map<string, number>();
   740→   740→    for (const n of notes) {
   741→   741→      for (const t of n.tags ?? []) {
   742→   742→        m.set(t.name, (m.get(t.name) ?? 0) + 1);
   743→   743→      }
   744→   744→    }
   745→   745→    return Array.from(m.entries()).sort((a, b) => b[1] - a[1]);
   746→   746→  }, [notes]);
   747→   747→
   748→   748→  if (tagMap.length === 0) {
   749→   749→    return (
   750→   750→      <p className="text-sm text-muted-foreground">No tags are in use yet.</p>
   751→   751→    );
   752→   752→  }
   753→   753→  return (
   754→   754→    <div className="flex flex-wrap gap-2">
   755→   755→      {tagMap.map(([name, count]) => (
   756→   756→        <Badge
   757→   757→          key={name}
   758→   758→          variant="outline"
   759→   759→          className="gap-1.5 py-1 font-normal"
   760→   760→        >
   761→   761→          <Tag className="size-3" />
   762→   762→          {name}
   763→   763→          <span className="text-xs text-muted-foreground">{count}</span>
   764→   764→        </Badge>
   765→   765→      ))}
   766→   766→    </div>
   767→   767→  );
   768→   768→}
