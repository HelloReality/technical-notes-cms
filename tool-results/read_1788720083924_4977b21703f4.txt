    92→// ─── Main NoteViewer ───────────────────────────────────────────
    93→export function NoteViewer() {
    94→  const note = useAppStore((s) => s.selectedNote);
    95→  const loading = useAppStore((s) => s.noteLoading);
    96→  const allNotes = useAppStore((s) => s.notes);
    97→  const openNote = useAppStore((s) => s.openNote);
    98→  const goHome = useAppStore((s) => s.goHome);
    99→
   100→  // State
   101→  const [toolbarExpanded, setToolbarExpanded] = React.useState(true);
   102→  const [drawerOpen, setDrawerOpen] = React.useState(false);
   103→  const [actionsOpen, setActionsOpen] = React.useState(false);
   104→  const [downloadOpen, setDownloadOpen] = React.useState(false);
   105→  const [zoomPresetsOpen, setZoomPresetsOpen] = React.useState(false);
   106→  const [pageViewMode, setPageViewMode] = React.useState<SidebarMode>("tiles");
   107→  const [isFullscreen, setIsFullscreen] = React.useState(false);
   108→  const [copied, setCopied] = React.useState(false);
   109→  const [zoom, setZoom] = React.useState(1);
   110→  const [zoomMode, setZoomMode] = React.useState<"fit" | "manual">("fit");
   111→  const [pageInputValue, setPageInputValue] = React.useState("");
   112→  const [pageInputFocused, setPageInputFocused] = React.useState(false);
   113→  const [isNavigating, setIsNavigating] = React.useState(false);
   114→
   115→  const containerRef = React.useRef<HTMLDivElement>(null);
   116→  const downloadRef = React.useRef<HTMLDivElement>(null);
   117→  const zoomRef = React.useRef<HTMLDivElement>(null);
   118→
   119→  // Pages
   120→  const pages = React.useMemo(() => {
   121→    if (!note) return [];
   122→    return allNotes
   123→      .filter((n) => n.categoryId === note.categoryId)
   124→      .sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   125→  }, [allNotes, note]);
   126→
   127→  const currentIndex = React.useMemo(() => {
   128→    if (!note) return -1;
   129→    return pages.findIndex((p) => p.id === note.id);
   130→  }, [pages, note]);
   131→
   132→  const totalPages = pages.length;
   133→  const prevPage = currentIndex > 0 ? pages[currentIndex - 1] : null;
   134→  const nextPage = currentIndex < totalPages - 1 ? pages[currentIndex + 1] : null;
   135→
   136→  const handleNavigate = React.useCallback((page: Note) => {
   137→    setIsNavigating(true);
   138→    openNote(page);
   139→    setDrawerOpen(false);
   140→    setTimeout(() => setIsNavigating(false), 400);
   141→  }, [openNote]);
   142→
   143→  // Fit-to-view
   144→  const computeFitScale = React.useCallback(() => {
   145→    const el = containerRef.current;
   146→    if (!el) return;
   147→    const w = el.clientWidth - 64;
   148→    if (w <= 0) return;
   149→    const fit = Math.min(w / 1136, 1);
   150→    setZoom(Math.max(0.25, fit));
   151→  }, []);
   152→
   153→  React.useEffect(() => {
   154→    if (zoomMode === "fit") {
   155→      const t = setTimeout(computeFitScale, 50);
   156→      return () => clearTimeout(t);
   157→    }
   158→  }, [note, zoomMode, computeFitScale, drawerOpen]);
   159→
   160→  React.useEffect(() => {
   161→    if (zoomMode !== "fit") return;
   162→    const handler = () => computeFitScale();
   163→    window.addEventListener("resize", handler);
   164→    return () => window.removeEventListener("resize", handler);
   165→  }, [zoomMode, computeFitScale]);
   166→
   167→  // Fullscreen
   168→  const toggleFullscreen = React.useCallback(() => {
   169→    if (!document.fullscreenElement) {
   170→      containerRef.current?.requestFullscreen?.();
   171→    } else {
   172→      document.exitFullscreen?.();
   173→    }
   174→  }, []);
   175→
   176→  React.useEffect(() => {
   177→    const h = () => setIsFullscreen(!!document.fullscreenElement);
   178→    document.addEventListener("fullscreenchange", h);
   179→    return () => document.removeEventListener("fullscreenchange", h);
   180→  }, []);
   181→
   182→  // Close popovers on outside click
   183→  React.useEffect(() => {
   184→    const handler = (e: MouseEvent) => {
   185→      if (downloadOpen && downloadRef.current && !downloadRef.current.contains(e.target as Node)) setDownloadOpen(false);
   186→      if (zoomPresetsOpen && zoomRef.current && !zoomRef.current.contains(e.target as Node)) setZoomPresetsOpen(false);
   187→    };
   188→    document.addEventListener("mousedown", handler);
   189→    return () => document.removeEventListener("mousedown", handler);
   190→  }, [downloadOpen, zoomPresetsOpen]);
   191→
   192→  // Keyboard
   193→  React.useEffect(() => {
   194→    const handleKey = (e: KeyboardEvent) => {
   195→      const target = e.target as HTMLElement;
   196→      const isTyping = target.tagName === "INPUT" || target.tagName === "TEXTAREA";
   197→
   198→      if (e.key === "Escape") {
   199→        if (drawerOpen) { setDrawerOpen(false); return; }
   200→        if (downloadOpen) { setDownloadOpen(false); return; }
   201→        if (actionsOpen) { setActionsOpen(false); return; }
   202→        if (zoomPresetsOpen) { setZoomPresetsOpen(false); return; }
   203→        setToolbarExpanded(false);
   204→      }
   205→
   206→      if ((e.ctrlKey || e.metaKey) && (e.key === "+" || e.key === "=")) {
   207→        e.preventDefault(); setZoomMode("manual"); setZoom(z => Math.min(4, z + 0.1)); return;
   208→      }
   209→      if ((e.ctrlKey || e.metaKey) && e.key === "-") {
   210→        e.preventDefault(); setZoomMode("manual"); setZoom(z => Math.max(0.25, z - 0.1)); return;
   211→      }
   212→      if ((e.ctrlKey || e.metaKey) && e.key === "0") {
   213→        e.preventDefault(); setZoomMode("fit"); computeFitScale(); return;
   214→      }
   215→      if (isTyping) return;
   216→      if (e.key === "ArrowLeft" && !drawerOpen && prevPage) { e.preventDefault(); handleNavigate(prevPage); }
   217→      if (e.key === "ArrowRight" && !drawerOpen && nextPage) { e.preventDefault(); handleNavigate(nextPage); }
   218→    };
   219→    window.addEventListener("keydown", handleKey);
   220→    return () => window.removeEventListener("keydown", handleKey);
   221→  }, [drawerOpen, downloadOpen, actionsOpen, zoomPresetsOpen, prevPage, nextPage, handleNavigate, computeFitScale]);
   222→
   223→  // Iframe load
   224→  const handleIframeLoad = React.useCallback((e: React.SyntheticEvent<HTMLIFrameElement>) => {
   225→    try {
   226→      const iframe = e.currentTarget;
   227→      const doc = iframe.contentDocument;
   228→      if (!doc) return;
   229→      const existing = doc.getElementById("reader-style");
   230→      if (existing) existing.remove();
   231→      const style = doc.createElement("style");
   232→      style.id = "reader-style";
   233→      style.textContent = `
   234→        body { justify-content: flex-start !important; padding-left: 42px !important; padding-right: 14px !important; }
   235→        @media (max-width: 1120px) { html { overflow-x: auto; } }
   236→      `;
   237→      doc.head.appendChild(style);
   238→      const resize = () => {
   239→        const h = Math.max(doc.body.scrollHeight, doc.body.offsetHeight, doc.documentElement.scrollHeight, doc.documentElement.offsetHeight);
   240→        iframe.style.height = `${h + 20}px`;
   241→      };
   242→      resize();
   243→      setTimeout(resize, 500);
   244→      setTimeout(resize, 1500);
   245→    } catch { /* sandbox */ }
   246→  }, []);
   247→
   248→  const handleShare = () => {
   249→    navigator.clipboard?.writeText(window.location.href).then(() => {
   250→      setCopied(true);
   251→      setTimeout(() => setCopied(false), 2000);
   252→    });
   253→  };
   254→
   255→  // Render
   256→  if (loading) {
   257→    return (
   258→      <div className="flex h-screen items-center justify-center bg-slate-800">
   259→        <motion.div
   260→          initial={{ opacity: 0, scale: 0.95 }}
   261→          animate={{ opacity: 1, scale: 1 }}
   262→          transition={{ duration: 0.3 }}
   263→        >
   264→          <Skeleton className="h-[60vh] w-full max-w-4xl rounded-xl" />
   265→        </motion.div>
   266→      </div>
   267→    );
   268→  }
   269→
   270→  if (!note) {
   271→    return (
   272→      <motion.div
   273→        initial={{ opacity: 0 }}
   274→        animate={{ opacity: 1 }}
   275→        className="flex h-screen flex-col items-center justify-center gap-4 bg-slate-800 text-center text-slate-200"
   276→      >
   277→        <motion.div
   278→          initial={{ scale: 0 }}
   279→          animate={{ scale: 1 }}
   280→          transition={{ type: "spring", damping: 20, stiffness: 200 }}
   281→          className="flex size-16 items-center justify-center rounded-full bg-slate-700"
   282→        >
   283→          <Home className="size-8 text-slate-400" />
   284→        </motion.div>
   285→        <div>
   286→          <h2 className="text-xl font-bold text-white">Note not found</h2>
   287→          <p className="mt-1 text-sm text-slate-400">The note doesn't exist or is no longer published.</p>
   288→        </div>
   289→        <Button onClick={goHome} className="gap-2">
   290→          <Home className="size-4" /> Browse all notes
   291→        </Button>
   292→      </motion.div>
   293→    );
   294→  }
   295→
   296→  const zoomLabel = zoomMode === "fit" ? "Fit" : `${Math.round(zoom * 100)}%`;
   297→
   298→  return (
   299→    <div ref={containerRef} className="relative h-screen w-full overflow-hidden bg-slate-800">
   300→      {/* ═══ Document Viewer ═══ */}
   301→      <div className="absolute inset-0 overflow-auto">
   302→        <div className="flex min-h-full items-start justify-center p-4 pt-16 sm:p-8 sm:pt-16">
   303→          <AnimatePresence mode="wait">
   304→            <motion.div
   305→              key={note.id}
   306→              initial={{ opacity: 0, y: 10 }}
   307→              animate={{ opacity: 1, y: 0 }}
   308→              exit={{ opacity: 0, y: -10 }}
   309→              transition={{ duration: 0.25, ease: "easeOut" }}
   310→              className="overflow-x-auto rounded-xl bg-white shadow-2xl"
   311→              style={{ maxWidth: `${1080 + 56}px`, width: "100%" }}
   312→            >
   313→              <iframe
   314→                title={note.title}
   315→                className="block"
   316→                style={{
   317→                  minHeight: "60vh",
   318→                  transform: `scale(${zoom})`,
   319→                  transformOrigin: "top left",
   320→                  width: `${100 / zoom}%`,
   321→                  border: 0,
   322→                }}
   323→                src={note.contentPath}
   324→                sandbox="allow-same-origin allow-popups"
   325→                onLoad={handleIframeLoad}
   326→              />
   327→            </motion.div>
   328→          </AnimatePresence>
   329→        </div>
   330→      </div>
   331→
   332→      {/* ═══ Top-left: Home + Breadcrumb ═══ */}
   333→      <motion.div
   334→        initial={{ opacity: 0, x: -10 }}
   335→        animate={{ opacity: 1, x: 0 }}
   336→        transition={{ duration: 0.3 }}
   337→        className="absolute left-3 top-3 z-30 flex items-center gap-2"
   338→      >
   339→        <motion.button
   340→          whileHover={{ scale: 1.05 }}
   341→          whileTap={{ scale: 0.95 }}
   342→          onClick={goHome}
   343→          className="flex h-9 w-9 items-center justify-center rounded-full border border-slate-200 bg-white/95 text-slate-600 shadow-md backdrop-blur-sm transition-colors hover:bg-white hover:text-slate-900"
   344→          title="Back to home"
   345→        >
   346→          <Home className="h-4 w-4" />
   347→        </motion.button>
   348→        <div className="hidden items-center gap-1.5 rounded-full border border-slate-200 bg-white/95 px-3 py-1.5 text-xs text-slate-500 shadow-md backdrop-blur-sm sm:flex">
   349→          {note.category && (
   350→            <>
   351→              <span className="text-slate-400">{note.category.name}</span>
   352→              <ChevronRight className="h-3 w-3 text-slate-300" />
   353→            </>
   354→          )}
   355→          <span className="max-w-[200px] truncate text-slate-700">{note.title}</span>
   356→          {totalPages > 1 && (
   357→            <span className="ml-1 text-slate-400">· {currentIndex + 1}/{totalPages}</span>
   358→          )}
   359→        </div>
   360→      </motion.div>
   361→
   362→      {/* ═══ Top-right: Pages + Actions ═══ */}
   363→      <motion.div
   364→        initial={{ opacity: 0, x: 10 }}
   365→        animate={{ opacity: 1, x: 0 }}
   366→        transition={{ duration: 0.3 }}
   367→        className="absolute right-3 top-3 z-30 flex items-center gap-2"
   368→      >
   369→        <motion.button
   370→          whileHover={{ scale: 1.05 }}
   371→          whileTap={{ scale: 0.95 }}
   372→          onClick={() => setDrawerOpen(true)}
   373→          className="flex h-9 w-9 items-center justify-center rounded-full border border-slate-200 bg-white/95 text-slate-600 shadow-md backdrop-blur-sm transition-colors hover:bg-white hover:text-slate-900"
   374→          title="Open pages"
   375→        >
   376→          <Menu className="h-4 w-4" />
   377→        </motion.button>
   378→        <motion.button
   379→          whileHover={{ scale: 1.05 }}
   380→          whileTap={{ scale: 0.95 }}
   381→          onClick={() => setActionsOpen(!actionsOpen)}
   382→          className={cn(
   383→            "flex h-9 w-9 items-center justify-center rounded-full border border-slate-200 shadow-md backdrop-blur-sm transition-colors",
   384→            actionsOpen ? "bg-white text-slate-900" : "bg-white/95 text-slate-600 hover:bg-white hover:text-slate-900"
   385→          )}
   386→          title="Toggle actions"
   387→        >
   388→          <MoreVertical className="h-4 w-4" />
   389→        </motion.button>
   390→      </motion.div>
   391→
   392→      {/* ═══ Top Floating Toolbar ═══ */}
   393→      <AnimatePresence>
   394→        {toolbarExpanded ? (
   395→          <motion.div
   396→            key="toolbar-expanded"
   397→            initial={{ opacity: 0, y: -20, scale: 0.9 }}
   398→            animate={{ opacity: 1, y: 0, scale: 1 }}
   399→            exit={{ opacity: 0, y: -20, scale: 0.9 }}
   400→            transition={{ type: "spring", damping: 25, stiffness: 300 }}
   401→            style={{ position: "absolute", left: "50%", top: "3rem", translateX: "-50%", zIndex: 20 }}
   402→          >
   403→            <div className="flex items-center gap-1 rounded-full border border-slate-200 bg-white/95 px-2 py-1.5 shadow-lg backdrop-blur-md">
   404→              {/* Download */}
   405→              <div className="relative" ref={downloadRef}>
   406→                <Button variant="ghost" size="icon"
   407→                  className={cn("h-8 w-8 rounded-full", downloadOpen && "bg-slate-100")}
   408→                  onClick={() => setDownloadOpen(!downloadOpen)} title="Export">
   409→                  <Download className="h-4 w-4" />
   410→                </Button>
   411→                <AnimatePresence>
   412→                  {downloadOpen && (
   413→                    <motion.div
   414→                      initial={{ opacity: 0, y: -5, scale: 0.95 }}
   415→                      animate={{ opacity: 1, y: 0, scale: 1 }}
   416→                      exit={{ opacity: 0, y: -5, scale: 0.95 }}
   417→                      transition={{ duration: 0.15 }}
   418→                      className="absolute left-0 top-full mt-2 w-56 rounded-lg border border-slate-200 bg-white p-1 shadow-xl"
   419→                    >
   420→                      <div className="px-2 py-1 text-[10px] font-bold uppercase tracking-wider text-slate-400">
   421→                        Current Page ({currentIndex + 1})
   422→                      </div>
   423→                      {["PNG", "JPEG", "PDF"].map(fmt => (
   424→                        <button key={fmt} onClick={() => setDownloadOpen(false)}
   425→                          className="flex w-full items-center gap-2 rounded-md px-2 py-1.5 text-xs text-slate-700 hover:bg-slate-100">
   426→                          <FileImage className="h-3.5 w-3.5 text-slate-400" /> {fmt}
   427→                        </button>
   428→                      ))}
   429→                      <div className="my-1 h-px bg-slate-100" />
   430→                      <div className="px-2 py-1 text-[10px] font-bold uppercase tracking-wider text-slate-400">All Pages</div>
   431→                      {["PDF", "PNG (ZIP)"].map(fmt => (
   432→                        <button key={fmt} onClick={() => setDownloadOpen(false)}
   433→                          className="flex w-full items-center gap-2 rounded-md px-2 py-1.5 text-xs text-slate-700 hover:bg-slate-100">
   434→                          <FileText className="h-3.5 w-3.5 text-slate-400" /> {fmt}
   435→                        </button>
   436→                      ))}
   437→                    </motion.div>
   438→                  )}
   439→                </AnimatePresence>
   440→              </div>
   441→
   442→              <Sep />
   443→
   444→              {/* Prev / page input / Next */}
   445→              <Button variant="ghost" size="icon" disabled={!prevPage}
   446→                onClick={() => prevPage && handleNavigate(prevPage)}
   447→                className="h-8 w-8 rounded-full" title="Previous page (←)">
   448→                <ChevronLeft className="h-4 w-4" />
   449→              </Button>
   450→              <div className="flex items-center gap-0.5 text-xs font-mono text-slate-700">
   451→                <input
   452→                  type="text"
   453→                  value={pageInputFocused ? pageInputValue : String(currentIndex + 1)}
   454→                  onChange={(e) => setPageInputValue(e.target.value.replace(/[^0-9]/g, ""))}
   455→                  onFocus={() => { setPageInputFocused(true); setPageInputValue(String(currentIndex + 1)); }}
   456→                  onBlur={() => {
   457→                    setPageInputFocused(false);
   458→                    const n = parseInt(pageInputValue, 10);
   459→                    if (!isNaN(n) && n >= 1 && n <= totalPages) handleNavigate(pages[n - 1]);
   460→                  }}
   461→                  onKeyDown={(e) => { if (e.key === "Enter") e.currentTarget.blur(); }}
   462→                  className="h-7 w-8 rounded border border-transparent bg-slate-100 text-center text-xs font-mono outline-none transition-colors hover:bg-slate-200 focus:border-slate-300 focus:bg-white"
   463→                  title="Type page number"
   464→                />
   465→                <span className="text-slate-400">/</span>
   466→                <span className="min-w-[20px] text-center">{totalPages}</span>
   467→              </div>
   468→              <Button variant="ghost" size="icon" disabled={!nextPage}
   469→                onClick={() => nextPage && handleNavigate(nextPage)}
   470→                className="h-8 w-8 rounded-full" title="Next page (→)">
   471→                <ChevronRight className="h-4 w-4" />
   472→              </Button>
   473→
   474→              <Sep />
   475→
   476→              {/* Zoom */}
   477→              <Button variant="ghost" size="icon" className="h-8 w-8 rounded-full"
   478→                onClick={() => { setZoomMode("manual"); setZoom(Math.max(0.25, zoom - 0.1)); }}
   479→                title="Zoom out (Ctrl -)">
   480→                <ZoomOut className="h-4 w-4" />
   481→              </Button>
   482→              <div className="relative" ref={zoomRef}>
   483→                <button
   484→                  className="h-7 min-w-[52px] rounded border border-transparent bg-slate-100 px-1.5 text-center text-xs font-mono transition-colors hover:border-slate-300 hover:bg-slate-200 outline-none"
   485→                  onClick={() => setZoomPresetsOpen(!zoomPresetsOpen)}
   486→                  title="Click to set zoom"
   487→                >
   488→                  {zoomLabel}
   489→                </button>
   490→                <AnimatePresence>
   491→                  {zoomPresetsOpen && (
   492→                    <motion.div
   493→                      initial={{ opacity: 0, y: -5, scale: 0.95 }}
   494→                      animate={{ opacity: 1, y: 0, scale: 1 }}
   495→                      exit={{ opacity: 0, y: -5, scale: 0.95 }}
   496→                      transition={{ duration: 0.15 }}
   497→                      className="absolute left-1/2 top-full mt-2 w-40 -translate-x-1/2 rounded-lg border border-slate-200 bg-white p-1 shadow-xl"
   498→                    >
   499→                      <button onClick={() => { setZoomMode("fit"); computeFitScale(); setZoomPresetsOpen(false); }}
   500→                        className="flex w-full items-center gap-2 rounded-md px-2 py-1.5 text-xs text-slate-700 hover:bg-slate-100">
   501→                        <Maximize className="h-3.5 w-3.5" /> Fit
   502→                      </button>
   503→                      <div className="my-1 h-px bg-slate-100" />
   504→                      {["50%", "75%", "100%", "125%", "150%", "200%"].map(label => (
   505→                        <button key={label} onClick={() => { setZoomMode("manual"); setZoom(parseInt(label) / 100); setZoomPresetsOpen(false); }}
   506→                          className="flex w-full items-center gap-2 rounded-md px-2 py-1.5 text-xs text-slate-700 hover:bg-slate-100">
   507→                          {label}
   508→                        </button>
   509→                      ))}
   510→                    </motion.div>
   511→                  )}
   512→                </AnimatePresence>
   513→              </div>
   514→              <Button variant="ghost" size="icon" className="h-8 w-8 rounded-full"
   515→                onClick={() => { setZoomMode("manual"); setZoom(Math.min(4, zoom + 0.1)); }}
   516→                title="Zoom in (Ctrl +)">
   517→                <ZoomIn className="h-4 w-4" />
   518→              </Button>
   519→
   520→              <Sep />
   521→
   522→              {/* Fullscreen */}
   523→              <Button variant="ghost" size="icon" className="h-8 w-8 rounded-full"
   524→                onClick={toggleFullscreen} title={isFullscreen ? "Exit fullscreen" : "Enter fullscreen"}>
   525→                {isFullscreen ? <Minimize className="h-4 w-4" /> : <Maximize className="h-4 w-4" />}
   526→              </Button>
   527→
   528→              {/* Collapse */}
   529→              <Button variant="ghost" size="icon" className="h-8 w-8 rounded-full"
   530→                onClick={() => setToolbarExpanded(false)} title="Collapse toolbar (Esc)">
   531→                <ChevronUp className="h-4 w-4" />
   532→              </Button>
   533→            </div>
   534→          </motion.div>
   535→        ) : (
   536→          <motion.button
   537→            key="toolbar-collapsed"
   538→            initial={{ opacity: 0, y: -10 }}
   539→            animate={{ opacity: 1, y: 0 }}
   540→            exit={{ opacity: 0, y: -10 }}
   541→            transition={{ delay: 0.1, duration: 0.2 }}
   542→            style={{ position: "absolute", left: "50%", top: "0.5rem", translateX: "-50%", zIndex: 20 }}
   543→            className="flex items-center gap-1.5 rounded-full border border-slate-200 bg-white/95 px-4 py-1.5 text-xs text-slate-600 shadow-md backdrop-blur-sm transition-colors hover:bg-white"
   544→            onClick={() => setToolbarExpanded(true)}
   545→            title="Expand toolbar"
   546→          >
   547→            <ChevronDown className="h-3 w-3" />
   548→            <span className="font-mono">{currentIndex + 1}/{totalPages}</span>
   549→          </motion.button>
   550→        )}
   551→      </AnimatePresence>
   552→
   553→      {/* ═══ Left Sidebar Drawer ═══ */}
   554→      <AnimatePresence>
   555→        {drawerOpen && (
   556→          <>
   557→            <motion.div
   558→              initial={{ opacity: 0 }}
   559→              animate={{ opacity: 1 }}
   560→              exit={{ opacity: 0 }}
   561→              transition={{ duration: 0.2 }}
   562→              className="absolute inset-0 z-30 bg-black/30"
   563→              onClick={() => setDrawerOpen(false)}
   564→            />
   565→            <motion.div
   566→              initial={{ x: "100%" }}
   567→              animate={{ x: 0 }}
   568→              exit={{ x: "100%" }}
   569→              transition={{ type: "spring", damping: 30, stiffness: 300 }}
   570→              className="absolute bottom-0 right-0 top-0 z-40 flex w-[85vw] flex-col bg-white shadow-2xl sm:w-80 sm:max-w-[320px]"
   571→            >
   572→              {/* Header */}
   573→              <div className="flex items-center justify-between border-b border-slate-200 px-4 py-3">
   574→                <span className="text-sm font-semibold text-slate-900">Pages</span>
   575→                <button onClick={() => setDrawerOpen(false)}
   576→                  className="rounded p-1 text-slate-400 hover:bg-slate-100 hover:text-slate-600">
   577→                  <X className="h-4 w-4" />
   578→                </button>
   579→              </div>
   580→              {/* Toggle */}
   581→              <div className="border-b border-slate-200 px-4 py-2">
   582→                <div className="flex w-fit items-center gap-0.5 rounded-md bg-slate-100 p-0.5">
   583→                  <button className={cn("rounded px-3 py-1 text-xs transition-all", pageViewMode === "list" ? "bg-white font-semibold shadow-sm" : "text-slate-500")}
   584→                    onClick={() => setPageViewMode("list")}>List</button>
   585→                  <button className={cn("rounded px-3 py-1 text-xs transition-all", pageViewMode === "tiles" ? "bg-white font-semibold shadow-sm" : "text-slate-500")}
   586→                    onClick={() => setPageViewMode("tiles")}>Tiles</button>
   587→                </div>
   588→              </div>
   589→              {/* Content */}
   590→              <ScrollArea className="flex-1">
   591→                {pageViewMode === "list" ? (
   592→                  <div className="space-y-0.5 p-3">
   593→                    {pages.map((page, idx) => (
   594→                      <motion.button
   595→                        key={page.id}
   596→                        custom={idx}
   597→                        initial={{ x: -10, opacity: 0 }}
   598→                        animate={{ x: 0, opacity: 1, transition: { delay: idx * 0.03, duration: 0.15, ease: "easeOut" } }}
   599→                        onClick={() => handleNavigate(page)}
   600→                        className={cn("flex w-full items-center rounded-lg border px-3 py-2.5 text-left",
   601→                          idx === currentIndex ? "border-rose-200 bg-rose-50" : "border-transparent hover:bg-slate-50")}
   602→                      >
   603→                        <span className={cn("mr-3 font-mono text-xs", idx === currentIndex ? "font-bold text-rose-600" : "text-slate-400")}>
   604→                          {String(idx + 1).padStart(2, "0")}
   605→                        </span>
   606→                        <span className={cn("truncate text-sm", idx === currentIndex ? "font-medium text-rose-900" : "text-slate-700")}>
   607→                          {page.title}
   608→                        </span>
   609→                      </motion.button>
   610→                    ))}
   611→                  </div>
   612→                ) : (
   613→                  <div className="grid grid-cols-2 gap-2 p-3">
   614→                    {pages.map((page, idx) => (
   615→                      <motion.button
   616→                        key={page.id}
   617→                        custom={idx}
   618→                        initial={{ opacity: 0, scale: 0.9 }}
   619→                        animate={{ opacity: 1, scale: 1, transition: { delay: idx * 0.04, duration: 0.2, ease: "easeOut" } }}
   620→                        whileHover={{ scale: 1.03, transition: { duration: 0.15 } }}
   621→                        whileTap={{ scale: 0.97 }}
   622→                        onClick={() => handleNavigate(page)}
   623→                        className={cn("overflow-hidden rounded-lg border-2 text-left",
   624→                          idx === currentIndex ? "border-rose-400 shadow-md" : "border-slate-200 hover:border-slate-300")}
   625→                      >
   626→                        <div className="border-b border-slate-100 bg-slate-50 px-2 py-1 text-center text-[10px] font-semibold">
   627→                          {String(idx + 1).padStart(2, "0")}
   628→                        </div>
   629→                        {/* Real preview */}
   630→                        <div className="relative aspect-[3/4] overflow-hidden bg-white">
   631→                          <iframe
   632→                            src={page.contentPath}
   633→                            className="pointer-events-none absolute left-0 top-0 border-0"
   634→                            style={{ width: "1080px", height: "1440px", transform: "scale(0.12)", transformOrigin: "top left" }}
   635→                            sandbox="allow-same-origin"
   636→                            title={page.title}
   637→                          />
   638→                        </div>
   639→                        <div className="truncate border-t border-slate-100 bg-slate-50 px-2 py-1 text-[10px] text-slate-600">
   640→                          {page.title}
   641→                        </div>
   642→                      </motion.button>
   643→                    ))}
   644→                  </div>
   645→                )}
   646→              </ScrollArea>
   647→            </motion.div>
   648→          </>
   649→        )}
   650→      </AnimatePresence>
   651→
   652→      {/* ═══ Right Actions Panel ═══ */}
   653→      <AnimatePresence>
   654→        {actionsOpen && (
   655→          <motion.div
   656→            initial={{ opacity: 0, x: 20, scale: 0.95 }}
   657→            animate={{ opacity: 1, x: 0, scale: 1 }}
   658→            exit={{ opacity: 0, x: 20, scale: 0.95 }}
   659→            transition={{ type: "spring", damping: 25, stiffness: 300 }}
   660→            className="absolute right-3 top-16 z-30 w-44 rounded-xl border border-slate-200 bg-white/95 p-3 shadow-lg backdrop-blur-md"
   661→          >
   662→            <div className="mb-2 flex items-center justify-between">
   663→              <span className="text-xs font-semibold uppercase tracking-wider text-slate-500">Actions</span>
   664→              <button onClick={() => setActionsOpen(false)} className="text-slate-400 hover:text-slate-600">
   665→                <X className="h-3.5 w-3.5" />
   666→              </button>
   667→            </div>
   668→            <div className="space-y-1">
   669→              <ActionBtn icon={Share2} label={copied ? "Link Copied!" : "Copy Link"} onClick={handleShare} />
   670→              <ActionBtn icon={ExternalLink} label="Open Raw" onClick={() => window.open(note.contentPath, "_blank")} />
   671→            </div>
   672→            <div className="mt-3 border-t border-slate-100 pt-3">
   673→              <div className="flex flex-col gap-1 text-xs text-slate-500">
   674→                <div className="flex items-center gap-1.5">
   675→                  <Clock className="h-3 w-3" /> Updated {fromNow(note.updatedAt)}
   676→                </div>
   677→                <div>{formatDateTime(note.updatedAt)}</div>
   678→              </div>
   679→              {note.tags && note.tags.length > 0 && (
   680→                <div className="mt-2 flex flex-wrap gap-1">
   681→                  {note.tags.map(t => (
   682→                    <Badge key={t.id} variant="outline" className="text-[10px] font-normal">{t.name}</Badge>
   683→                  ))}
   684→                </div>
   685→              )}
   686→            </div>
   687→          </motion.div>
   688→        )}
   689→      </AnimatePresence>
   690→    </div>
   691→  );
   692→}
