    67→# Helper HTML builders
    68→# ============================================================
    69→def row(icon_name: str, bg: str, ic_color: str, heading: str, body_html: str) -> str:
    70→    """Build a `.row` block with icon + heading + body text."""
    71→    return (
    72→        f'<div class="row">'
    73→        f'<div class="icon {bg} {ic_color}">{svg(icon_name)}</div>'
    74→        f'<div class="text"><h3>{heading}</h3>{body_html}</div>'
    75→        f'</div>'
    76→    )
    77→
    78→
    79→def terminal(title: str, body_html: str) -> str:
    80→    """Build a `.terminal` block with macOS-style title bar."""
    81→    return (
    82→        f'<div class="terminal">'
    83→        f'<div class="term-bar">'
    84→        f'<span class="term-dot dot-red"></span>'
    85→        f'<span class="term-dot dot-amber"></span>'
    86→        f'<span class="term-dot dot-green"></span>'
    87→        f'<span class="term-title">{title}</span>'
    88→        f'</div>'
    89→        f'<div class="term-body">{body_html}</div>'
    90→        f'</div>'
    91→    )
    92→
    93→
    94→def card(head_icon: str, head_text: str, body_html: str) -> str:
    95→    """Build a `.card` sidebar item."""
    96→    return (
    97→        f'<div class="card">'
    98→        f'<div class="head">{svg(head_icon)}<span>{head_text}</span></div>'
    99→        f'<div class="body-pad">{body_html}</div>'
   100→        f'</div>'
   101→    )
   102→
   103→
   104→def callout(kind: str, ico: str, text_html: str) -> str:
   105→    """Build a `.callout` (warn/tip) box."""
   106→    return (
   107→        f'<div class="callout {kind}">'
   108→        f'<span class="ico {("ic-orange" if kind == "warn" else "ic-green")}">{svg(ico)}</span>'
   109→        f'<div>{text_html}</div>'
   110→        f'</div>'
   111→    )
