   216→    """Wrap page content in a full HTML document with canonical + shared + page CSS."""

   222→<title>{title}</title>
   223→<style>
   224→/* ====== CANONICAL NOTEBOOK TEMPLATE CSS ====== */
   225→{CANONICAL_CSS}

   227→/* ====== SHARED PAGE EXTENSIONS ====== */
   228→{SHARED_EXTRA_CSS}
   229→
   230→/* ====== PAGE-SPECIFIC STYLES ====== */
   231→{page_css}
   232→</style>
   233→</head>
   234→<body class="multi-page">
   235→<div class="page-wrapper">
   236→  <div class="page">
   237→    <div class="spiral"></div>
   238→    <div class="holes"></div>
   239→    <div class="page-bend"></div>
   240→    <div class="page-top-edge"></div>
   241→    <div class="page-bottom-edge"></div>
   242→    <div class="content">

   250→
   251→def head(brand_text: str, page_label: str, title_html: str, subtitle_html: str = "") -> str:
   252→    """Page header (brand + badge + title + subtitle)."""
   253→    sub = ""
   254→    if subtitle_html:
   255→        sub = f'<div class="page-subtitle">{subtitle_html}</div>'
   256→    else:
   257→        sub = '<div class="page-subtitle">&nbsp;</div>'
   258→    return f"""
   259→<div class="page-head">
   260→  <div class="brand"><span class="vq">S</span>{brand_text}</div>
   261→  <div class="badge">{page_label}</div>
   262→</div>
   263→<h1 class="page-title">{title_html}</h1>
   264→{sub}
   265→"""
   266→
   267→
   268→def card(head_html: str, body_html: str, extra_cls: str = "") -> str:
   269→    """A standard .card with header + pad body."""
   270→    return f'<div class="card {extra_cls}"><div class="head">{head_html}</div><div class="pad">{body_html}</div></div>'
   271→
   272→
   273→def terminal_block(title: str, body_lines: list, extra_cls: str = "") -> str:
   274→    """Dark terminal block. body_lines is a list of (cls, text)."""
   275→    dots = (
   276→        '<span class="dot dot-r"></span>'
   277→        '<span class="dot dot-y"></span>'
   278→        '<span class="dot dot-g"></span>'
   279→    )
   280→    body_html = "\n".join(
   281→        f'<div class="{cls}">{txt}</div>' if cls else f'<div>{txt}</div>'
   282→        for cls, txt in body_lines
   283→    )
   284→    return f"""<div class="terminal {extra_cls}">
   285→  <div class="bar">{dots}<span class="ttl">{title}</span></div>
   286→  <div class="body">{body_html}</div>
   287→</div>"""
