    50→    50→function buildTree(categories: Category[], notes: Note[]): TreeNode[] {
    51→    51→  const top = categories.filter((c) => !c.parentId);
    52→    52→  return top
    53→    53→    .map((cat) => {
    54→    54→      const topics = categories
    55→    55→        .filter((c) => c.parentId === cat.id)
    56→    56→        .map((sub) => ({
    57→    57→          category: sub,
    58→    58→          notes: notes
    59→    59→            .filter((n) => n.categoryId === sub.id)
    60→    60→            .sort(
    61→    61→              (a, b) =>
    62→    62→                new Date(a.createdAt).getTime() -
    63→    63→                new Date(b.createdAt).getTime(),
    64→    64→            ),
    65→    65→        }))
    66→    66→        .filter((t) => t.notes.length > 0);
    67→    67→      const directNotes = notes
    68→    68→        .filter((n) => n.categoryId === cat.id)
    69→    69→        .sort(
    70→    70→          (a, b) =>
    71→    71→            new Date(a.createdAt).getTime() -
    72→    72→            new Date(b.createdAt).getTime(),
    73→    73→        );
    74→    74→      return { category: cat, topics, notes: directNotes };
    75→    75→    })
    76→    76→    .filter((n) => n.topics.length > 0 || n.notes.length > 0);
    77→    77→}
