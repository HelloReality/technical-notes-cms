    94→    94→function buildTree(categories: Category[], notes: Note[]): TreeNode[] {
    95→    95→  const top = categories.filter((c) => !c.parentId);
    96→    96→  return top
    97→    97→    .map((cat) => {
    98→    98→      const topics = categories
    99→    99→        .filter((c) => c.parentId === cat.id)
   100→   100→        .map((sub) => ({
   101→   101→          category: sub,
   102→   102→          notes: notes
   103→   103→            .filter((n) => n.categoryId === sub.id)
   104→   104→            .sort(
   105→   105→              (a, b) =>
   106→   106→                new Date(a.createdAt).getTime() -
   107→   107→                new Date(b.createdAt).getTime(),
   108→   108→            ),
   109→   109→        }))
   110→   110→        .filter((t) => t.notes.length > 0);
   111→   111→      const directNotes = notes
   112→   112→        .filter((n) => n.categoryId === cat.id)
   113→   113→        .sort(
   114→   114→          (a, b) =>
   115→   115→            new Date(a.createdAt).getTime() -
   116→   116→            new Date(b.createdAt).getTime(),
   117→   117→        );
   118→   118→      return { category: cat, topics, notes: directNotes };
   119→   119→    })
   120→   120→    .filter((n) => n.topics.length > 0 || n.notes.length > 0);
   121→   121→}

   324→   324→  // margin — is treated as ONE renderable canvas. We measure the

   334→   334→  const applyNoteScale = React.useCallback(() => {

   340→   340→      if (!doc || !doc.body) return;
   341→   341→
   342→   342→      // Neutralize the body's outer padding so the page-wrapper centers

   344→   344→      // design — only removes the decorative margin the raw file uses
   345→   345→      // when viewed standalone.
   346→   346→      const NEUTRALIZE_ID = "reader-neutralize-body-pad";

   358→   358→      const VIEWPORT_MARGIN = 16;

   360→   360→

   362→   362→      const pageEl =
   363→   363→        (doc.querySelector(".page-wrapper") as HTMLElement | null) ||

   367→   367→        doc.body;

   371→   371→      // the iframe's current laid-out width, avoiding chicken-and-egg
   372→   372→      // measurement issues.

   374→   374→      const parsedW = computedW ? parseFloat(computedW) : 0;
   375→   375→      const pageWNum = isFinite(parsedW) && parsedW > 0 ? parsedW : 1080;
   376→   376→
   377→   377→      // The complete page object width = page + ring overflow (left) +
   378→   378→      // shadow spread (both sides). The rings extend ~40px left of the
   379→   379→      // page; the box-shadow spreads ~60px on all sides.
   380→   380→      const RING_OVERFLOW = 40;
   381→   381→      const SHADOW = 60;
   382→   382→      const objectW = pageWNum + RING_OVERFLOW + SHADOW * 2;
   383→   383→
   384→   384→      // Body height includes the page; add shadow for the bottom.
   385→   385→      const bodyH = Math.max(
   386→   386→        doc.body.scrollHeight,
   387→   387→        doc.body.offsetHeight,
   388→   388→        doc.documentElement.scrollHeight,
   389→   389→        doc.documentElement.offsetHeight,
   390→   390→      );
   391→   391→      const objectH = bodyH + SHADOW;
   392→   392→
   393→   393→      // Base fit-to-width scale (page object fills the viewport).
   394→   394→      const fitScale = targetW < objectW ? targetW / objectW : 1;
   395→   395→      // Apply the user's zoom multiplier. In "fit" mode zoom stays at 1
   396→   396→      // (fitScale alone). In "manual" mode the user's zoom (0.25–4)
   397→   397→      // scales relative to the fit width.
   398→   398→      const zoomMul = zoomMode === "manual" ? zoom : 1;
   399→   399→      const scale = fitScale * zoomMul;
   400→   400→
   401→   401→      // Size the iframe to the complete page object.
   402→   402→      const NATIVE_W = objectW;
   403→   403→      iframe.style.width = `${NATIVE_W}px`;
   404→   404→      iframe.style.height = `${objectH}px`;
   405→   405→      iframe.style.transform = `scale(${scale})`;
   406→   406→      iframe.style.transformOrigin = "top left";
   407→   407→
   408→   408→      // Reserve the scaled footprint on the wrapper so the page can
   409→   409→      // scroll vertically and centers horizontally. When zoomed in
   410→   410→      // beyond the viewport width, the wrapper grows wider than the
   411→   411→      // container and the scroll container handles horizontal scroll.
   412→   412→      const wrapper = iframe.parentElement;
   413→   413→      if (wrapper) {
   414→   414→        wrapper.style.width = `${NATIVE_W * scale}px`;
   415→   415→        wrapper.style.height = `${objectH * scale}px`;
   416→   416→        wrapper.style.marginLeft = "auto";
   417→   417→        wrapper.style.marginRight = "auto";
   418→   418→      }
   419→   419→    } catch {
   420→   420→      /* cross-origin — ignore */
   421→   421→    }
   422→   422→  }, [zoom, zoomMode]);
   423→   423→

   435→   435→
   436→   436→  // Re-apply the scale whenever the user changes zoom / zoom mode.
