   608→  const binding = document.getElementById("binding");
   609→  if (!binding) return;
   610→  const NS = "http://www.w3.org/2000/svg";
   611→
   612→  function make(tag, attrs = {}) {
   613→    const el = document.createElementNS(NS, tag);
   614→    for (const [key, value] of Object.entries(attrs)) {
   615→      el.setAttribute(key, value);
   616→    }
   617→    return el;
   618→  }
   619→
   620→  function buildBinding() {
   621→    binding.innerHTML = "";
   622→
   623→    const width = binding.clientWidth || 100;
   624→    const height = binding.clientHeight || 1400;
   625→
   626→    /* ============================================================
   627→       U-SHAPED HAIRPIN DOUBLE-WIRE BINDING (matches reference)
   628→       Each unit = ONE continuous black wire bent into a shallow U:
   629→         - Two horizontal strands
   630→         - Rounded 180° curve on the LEFT connecting them
   631→         - Both right ends enter a small black connector
   632→       Built with SVG <path> (NOT <ellipse>, NOT <rect> wires).
   633→       ============================================================ */
   634→
   635→    /* Geometry — compact double-wire loop with clearly visible rounded bend */
   636→    const wireLeft       = 2;        // leftmost x of the bend (apex of the curve)
   637→    const wireRight      = 60;       // right end of wires (enters connector)
   638→    const wireStroke     = 3.5;      // wire thickness (3-4px per spec)
   639→    const wireGap        = 6;        // vertical gap between the two strands (5-7px per spec)
   640→    const wireColor      = "#080808"; // near-black, matte
   641→    const bendWidth      = 28;       // width of the left rounded bend (wide enough to be clearly visible)
   642→
   643→    /* Connector — small rounded VERTICAL rectangle (NOT a circle/dot) */
   644→    const connectorX     = 64;       // connector left (slight overlap with wire ends)
   645→    const connectorW     = 15;       // connector width (14-18px per spec)
   646→    const connectorH     = 17;       // connector height (14-18px per spec)
   647→    const connectorRx     = 3;        // slightly rounded corners (clearly a rectangle, not a circle)
   648→
   649→    /* Vertical rhythm */
   650→    const pitch  = Math.max(42, Math.min(54, height / 20));
   651→    const start  = 30;
   652→    const count  = Math.ceil((height - start + 20) / pitch);
   653→
   654→    const svg = make("svg", {
   655→      viewBox: `0 0 ${width} ${height}`,
   656→      width: width,
   657→      height: height,
   658→      preserveAspectRatio: "none"  /* Workaround: will be removed after appending to fix rendering */
   659→    });
   660→
   661→    const defs = make("defs");
   662→
   663→    /* Subtle wire gradient — near-black with very subtle top highlight */
   664→    const wireGrad = make("linearGradient", { id:"wireGrad", x1:"0%", y1:"0%", x2:"0%", y2:"100%" });
   665→    wireGrad.appendChild(make("stop",{offset:"0%","stop-color":"#1a1a1a"}));    /* subtle top */
   666→    wireGrad.appendChild(make("stop",{offset:"40%","stop-color":"#080808"}));    /* body */
   667→    wireGrad.appendChild(make("stop",{offset:"100%","stop-color":"#000000"}));   /* core */
   668→    defs.appendChild(wireGrad);
   669→
   670→    /* Connector gradient — slightly 3D rounded cap */
   671→    const connectorGrad = make("linearGradient", { id:"connectorGrad", x1:"0%", y1:"0%", x2:"100%", y2:"0%" });
   672→    connectorGrad.appendChild(make("stop",{offset:"0%","stop-color":"#1a1a1a"}));
   673→    connectorGrad.appendChild(make("stop",{offset:"40%","stop-color":"#050505"}));
   674→    connectorGrad.appendChild(make("stop",{offset:"100%","stop-color":"#000000"}));
   675→    defs.appendChild(connectorGrad);
   676→

   678→    const pageShadowGrad = make("linearGradient", { id:"pageEdgeShadow", x1:"0%", y1:"0%", x2:"100%", y2:"0%" });
   679→    pageShadowGrad.appendChild(make("stop",{offset:"0%","stop-color":"#000000","stop-opacity":"0.16"}));
   680→    pageShadowGrad.appendChild(make("stop",{offset:"40%","stop-color":"#000000","stop-opacity":"0.05"}));
   681→    pageShadowGrad.appendChild(make("stop",{offset:"100%","stop-color":"#000000","stop-opacity":"0"}));
   682→    defs.appendChild(pageShadowGrad);
   683→
   684→    svg.appendChild(defs);
   685→
   686→    /* ---- Page-edge vertical shadow (drawn once, full height) ---- */
   687→    svg.appendChild(make("rect",{
   688→      x:connectorX + connectorW - 2, y:0, width:12, height:height,
   689→      fill:"url(#pageEdgeShadow)"
   690→    }));
   691→

   693→    for (let i = 0; i < count; i++) {
   694→      const cy = start + i * pitch;
   695→
   696→      /* The two horizontal wire strands — vertically centered around cy */
   697→      const wire1Y = cy - wireGap/2;   // upper strand Y
   698→      const wire2Y = cy + wireGap/2;   // lower strand Y
   699→      const bendStartX = wireLeft + bendWidth;   // x where the bend begins
   700→
   701→      /* === LAYER 1: U-shaped wire — drawn as SEPARATE elements for reliable rendering ===
   702→         1. Upper horizontal wire (line)
   703→         2. Lower horizontal wire (line)
   704→         3. Left rounded bend (Bézier path connecting the two wires)
   705→         All solid black (#080808), same stroke width. */
   706→
   707→      /* Upper horizontal wire — thin rect for reliable rendering */
   708→      svg.appendChild(make("rect",{
   709→        x:bendStartX, y:wire1Y - wireStroke/2,
   710→        width:wireRight - bendStartX, height:wireStroke,
   711→        rx:wireStroke/2, fill:"#080808"
   712→      }));
   713→      /* Lower horizontal wire — thin rect for reliable rendering */
   714→      svg.appendChild(make("rect",{
   715→        x:bendStartX, y:wire2Y - wireStroke/2,
   716→        width:wireRight - bendStartX, height:wireStroke,
   717→        rx:wireStroke/2, fill:"#080808"
   718→      }));
   719→      /* Left rounded bend — Bézier connecting the two wire ends */
   720→      svg.appendChild(make("path",{
   721→        d:`M ${bendStartX} ${wire1Y} C ${wireLeft} ${wire1Y}, ${wireLeft} ${wire2Y}, ${bendStartX} ${wire2Y}`,
   722→        fill:"none", stroke:"#080808", "stroke-width":wireStroke,
   723→        "stroke-linecap":"round", "stroke-linejoin":"round"
   724→      }));
   725→
   726→      /* === LAYER 2: Very subtle top-edge highlight (matte, not shiny) ===
   727→         Thin lighter line along the upper edge of the upper strand only. */
   728→      svg.appendChild(make("line",{
   729→        x1:bendStartX + 1, y1:wire1Y - wireStroke/2 + 0.5,
   730→        x2:wireRight - 2, y2:wire1Y - wireStroke/2 + 0.5,
   731→        stroke:"#3a3a3a", "stroke-width":"0.5",
   732→        "stroke-linecap":"round", opacity:"0.45"
   733→      }));
   734→
   735→      /* === LAYER 3: Rounded black connector (FRONT — covers wire ends) ===
   736→         Single shared cap that both wire ends enter.
   737→         Small, rounded, NOT a perfect circle. */
   738→      /* Connector shadow (subtle, behind, offset down-right) */

   740→        x:connectorX, y:cy - connectorH/2, width:connectorW, height:connectorH,
   741→        rx:connectorRx, fill:"#000", opacity:"0.35",
   742→        transform:"translate(0.8 1.2)"

   744→      /* Main connector body — solid black */
   745→      svg.appendChild(make("rect",{
