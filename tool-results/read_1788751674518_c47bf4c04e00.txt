    41→// ─── Tree node type ───────────────────────────────────────────
    42→interface TreeNode {
    43→  category: Category;

    47→
    48→function buildTree(categories: Category[], notes: Note[]): TreeNode[] {
    49→  const top = categories.filter((c) => !c.parentId);
    50→  return top
    51→    .map((cat) => {
    52→      const topics = categories
    53→        .filter((c) => c.parentId === cat.id)
    54→        .map((sub) => ({
    55→          category: sub,
    56→          notes: notes
    57→            .filter((n) => n.categoryId === sub.id)
    58→            .sort(
    59→              (a, b) =>
    60→                new Date(a.createdAt).getTime() -
    61→                new Date(b.createdAt).getTime(),
    62→            ),
    63→        }))
    64→        .filter((t) => t.notes.length > 0);
    65→      const directNotes = notes
    66→        .filter((n) => n.categoryId === cat.id)
    67→        .sort(
    68→          (a, b) =>
    69→            new Date(a.createdAt).getTime() -
    70→            new Date(b.createdAt).getTime(),
    71→        );
    72→      return { category: cat, topics, notes: directNotes };
    73→    })
    74→    .filter((n) => n.topics.length > 0 || n.notes.length > 0);
    75→}

   144→  const [drawerOpen, setDrawerOpen] = React.useState(false);

   147→  const [zoomPresetsOpen, setZoomPresetsOpen] = React.useState(false);
   148→  const [pageViewMode, setPageViewMode] = React.useState<SidebarMode>("tiles");

   150→  const [copied, setCopied] = React.useState(false);
   151→  const [zoom, setZoom] = React.useState(1);
   152→  const [zoomMode, setZoomMode] = React.useState<"fit" | "manual">("fit");
   153→  const [pageInputValue, setPageInputValue] = React.useState("");
   154→  const [pageInputFocused, setPageInputFocused] = React.useState(false);
   155→  const [isNavigating, setIsNavigating] = React.useState(false);
   156→
   157→  // Tree drawer state (Category → Topic → subtopic navigation)
   158→  const [treeOpen, setTreeOpen] = React.useState(false);
   159→  const [expandedCats, setExpandedCats] = React.useState<Set<string>>(new Set());
   160→  const [expandedTopics, setExpandedTopics] = React.useState<Set<string>>(new Set());
   161→
   162→  const containerRef = React.useRef<HTMLDivElement>(null);
   163→  const mainRef = React.useRef<HTMLDivElement>(null);
   164→  const downloadRef = React.useRef<HTMLDivElement>(null);
   165→  const zoomRef = React.useRef<HTMLDivElement>(null);
   166→
   167→  // ─── Build tree ─────────────────────────────────────────────
   168→  const tree = React.useMemo(

   177→    if (!noteCat) return;

   180→    if (noteCat.parentId)
   181→      setExpandedTopics((prev) => new Set([...prev, noteCat.id]));

   183→

   185→  const pages = React.useMemo(() => {
   186→    if (!note) return [];

   188→      .filter((n) => n.categoryId === note.categoryId)
   189→      .sort((a, b) => new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime());
   190→  }, [allNotes, note]);
   191→
   192→  const currentIndex = React.useMemo(() => {
   193→    if (!note) return -1;
   194→    return pages.findIndex((p) => p.id === note.id);
   195→  }, [pages, note]);
   196→
   197→  const totalPages = pages.length;
   198→  const prevPage = currentIndex > 0 ? pages[currentIndex - 1] : null;
   199→  const nextPage = currentIndex < totalPages - 1 ? pages[currentIndex + 1] : null;
   200→
   201→  const handleNavigate = React.useCallback((page: Note) => {
   202→    setIsNavigating(true);
   203→    openNote(page);
   204→    setDrawerOpen(false);
   205→    setTreeOpen(false);
   206→    setTimeout(() => setIsNavigating(false), 400);
   207→  }, [openNote]);
   208→
   209→  const toggleCat = (id: string) =>
   210→    setExpandedCats((prev) => {
   211→      const n = new Set(prev);
   212→      if (n.has(id)) n.delete(id);
   213→      else n.add(id);
   214→      return n;
   215→    });
   216→  const toggleTopic = (id: string) =>
   217→    setExpandedTopics((prev) => {
   218→      const n = new Set(prev);
   219→      if (n.has(id)) n.delete(id);
   220→      else n.add(id);
   221→      return n;
   222→    });
   223→

   229→    if (w <= 0) return;
   230→    const fit = Math.min(w / 1136, 1);
   231→    setZoom(Math.max(0.25, fit));
   232→  }, []);
   233→
   234→  React.useEffect(() => {
   235→    if (zoomMode === "fit") {
   236→      const t = setTimeout(computeFitScale, 50);
   237→      return () => clearTimeout(t);
   238→    }
   239→  }, [note, zoomMode, computeFitScale, drawerOpen, treeOpen]);
   240→
   241→  React.useEffect(() => {
   242→    if (zoomMode !== "fit") return;
   243→    const handler = () => computeFitScale();
   244→    window.addEventListener("resize", handler);
   245→    return () => window.removeEventListener("resize", handler);
   246→  }, [zoomMode, computeFitScale]);
   247→
   248→  // Fullscreen
   249→  const toggleFullscreen = React.useCallback(() => {
   250→    if (!document.fullscreenElement) {
   251→      containerRef.current?.requestFullscreen?.();
   252→    } else {
   253→      document.exitFullscreen?.();

   255→  }, []);
   256→
   257→  React.useEffect(() => {
   258→    const h = () => setIsFullscreen(!!document.fullscreenElement);
   259→    document.addEventListener("fullscreenchange", h);
   260→    return () => document.removeEventListener("fullscreenchange", h);
   261→  }, []);
   262→
   263→  // Close popovers on outside click
   264→  React.useEffect(() => {
   265→    const handler = (e: MouseEvent) => {
   266→      if (downloadOpen && downloadRef.current && !downloadRef.current.contains(e.target as Node)) setDownloadOpen(false);
   267→      if (zoomPresetsOpen && zoomRef.current && !zoomRef.current.contains(e.target as Node)) setZoomPresetsOpen(false);
   268→    };
   269→    document.addEventListener("mousedown", handler);
   270→    return () => document.removeEventListener("mousedown", handler);
   271→  }, [downloadOpen, zoomPresetsOpen]);
   272→
   273→  // Keyboard
   274→  React.useEffect(() => {
   275→    const handleKey = (e: KeyboardEvent) => {
   276→      const target = e.target as HTMLElement;
   277→      const isTyping = target.tagName === "INPUT" || target.tagName === "TEXTAREA";
   278→
   279→      if (e.key === "Escape") {
   280→        if (treeOpen) { setTreeOpen(false); return; }
   281→        if (drawerOpen) { setDrawerOpen(false); return; }
   282→        if (downloadOpen) { setDownloadOpen(false); return; }
   283→        if (actionsOpen) { setActionsOpen(false); return; }
   284→        if (zoomPresetsOpen) { setZoomPresetsOpen(false); return; }
   285→        setToolbarExpanded(false);
   286→      }
   287→
   288→      if ((e.ctrlKey || e.metaKey) && (e.key === "+" || e.key === "=")) {
   289→        e.preventDefault(); setZoomMode("manual"); setZoom(z => Math.min(4, z + 0.1)); return;
   290→      }
   291→      if ((e.ctrlKey || e.metaKey) && e.key === "-") {
   292→        e.preventDefault(); setZoomMode("manual"); setZoom(z => Math.max(0.25, z - 0.1)); return;
   293→      }
   294→      if ((e.ctrlKey || e.metaKey) && e.key === "0") {
   295→        e.preventDefault(); setZoomMode("fit"); computeFitScale(); return;
   296→      }
   297→      if (isTyping) return;
   298→      if (e.key === "ArrowLeft" && !drawerOpen && !treeOpen && prevPage) { e.preventDefault(); handleNavigate(prevPage); }
   299→      if (e.key === "ArrowRight" && !drawerOpen && !treeOpen && nextPage) { e.preventDefault(); handleNavigate(nextPage); }
   300→    };
   301→    window.addEventListener("keydown", handleKey);
   302→    return () => window.removeEventListener("keydown", handleKey);
   303→  }, [drawerOpen, treeOpen, downloadOpen, actionsOpen, zoomPresetsOpen, prevPage, nextPage, handleNavigate, computeFitScale]);
   304→
   305→  const handleShare = () => {
   306→    navigator.clipboard?.writeText(window.location.href).then(() => {
   307→      setCopied(true);
   308→      setTimeout(() => setCopied(false), 2000);
   309→    });
   310→  };
   311→
   312→  // Render
   313→  if (loading) {
   314→    return (
   315→      <div className="flex h-screen items-center justify-center bg-slate-800">
   316→        <motion.div
   317→          initial={{ opacity: 0, scale: 0.95 }}
   318→          animate={{ opacity: 1, scale: 1 }}
   319→          transition={{ duration: 0.3 }}
   320→        >
   321→          <Skeleton className="h-[60vh] w-full max-w-4xl rounded-xl" />
   322→        </motion.div>
   323→      </div>
   324→    );
   325→  }
   326→
   327→  if (!note) {
   328→    return (
   329→      <motion.div
   330→        initial={{ opacity: 0 }}
   331→        animate={{ opacity: 1 }}
   332→        className="flex h-screen flex-col items-center justify-center gap-4 bg-slate-800 text-center text-slate-200"
   333→      >
   334→        <motion.div
   335→          initial={{ scale: 0 }}
   336→          animate={{ scale: 1 }}
   337→          transition={{ type: "spring", damping: 20, stiffness: 200 }}
   338→          className="flex size-16 items-center justify-center rounded-full bg-slate-700"
   339→        >
   340→          <Home className="size-8 text-slate-400" />
   341→        </motion.div>
   342→        <div>
   343→          <h2 className="text-xl font-bold text-white">Note not found</h2>
   344→          <p className="mt-1 text-sm text-slate-400">The note doesn't exist or is no longer published.</p>
   345→        </div>
   346→        <Button onClick={goHome} className="gap-2">
   347→          <Home className="size-4" /> Browse all notes
   348→        </Button>
   349→      </motion.div>
   350→    );
   351→  }
   352→
   353→  const zoomLabel = zoomMode === "fit" ? "Fit" : `${Math.round(zoom * 100)}%`;
   354→
   355→  return (

   361→        initial={false}
   362→        animate={{ width: treeOpen ? 248 : 48 }}

   364→        className="relative z-30 shrink-0 overflow-hidden border-r border-slate-200 bg-slate-50"

   366→        <div className="flex h-full w-[248px]">

   380→              <FolderTree className="h-4 w-4" />

   382→            <button
   383→              onClick={goHome}

   436→                  const isActiveCat = allNotes.some((n) => n.id === note.id);
