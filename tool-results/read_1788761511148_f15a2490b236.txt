    88→interface TreeNode {

    94→function buildTree(categories: Category[], notes: Note[]): TreeNode[] {
    95→  const top = categories.filter((c) => !c.parentId);
    96→  return top
    97→    .map((cat) => {
    98→      const topics = categories
    99→        .filter((c) => c.parentId === cat.id)
   100→        .map((sub) => ({
   101→          category: sub,
   102→          notes: notes
   103→            .filter((n) => n.categoryId === sub.id)
   104→            .sort(
   105→              (a, b) =>
   106→                new Date(a.createdAt).getTime() -
   107→                new Date(b.createdAt).getTime(),
   108→            ),
   109→        }))
   110→        .filter((t) => t.notes.length > 0);
   111→      const directNotes = notes
   112→        .filter((n) => n.categoryId === cat.id)
   113→        .sort(
   114→          (a, b) =>
   115→            new Date(a.createdAt).getTime() -
   116→            new Date(b.createdAt).getTime(),
   117→        );
   118→      return { category: cat, topics, notes: directNotes };
   119→    })
   120→    .filter((n) => n.topics.length > 0 || n.notes.length > 0);
   121→}

   321→  // ─── Note scaling (unified page object) ───────────────────────
   322→  // The ENTIRE notebook sheet — spiral rings + ring shadows + page
   323→  // shadow + the complete page design + a small intentional outer
   324→  // margin — is treated as ONE renderable canvas. We measure the
   325→  // note's true bounding box (including absolutely-positioned rings
   326→  // and the box-shadow overflow), size the iframe to that exact box,
   327→  // then scale the whole object as a unit to fit the viewport width.
   328→  //
   329→  // The note's internal design is NEVER modified — we only neutralize
   330→  // the body's outer padding (which is decorative spacing for the raw
   331→  // standalone file) so the page centers cleanly in the iframe. No
   332→  // reflow, no redesign. Only the scale of the complete page object
   333→  // adjusts responsively per device.
   334→  const applyNoteScale = React.useCallback(() => {
   335→    const iframe = iframeRef.current;
   336→    const container = mainRef.current;
   337→    if (!iframe || !container) return;
   338→    try {
   339→      const doc = iframe.contentDocument;
   340→      if (!doc || !doc.body) return;
   341→
   342→      // Neutralize the body's outer padding so the page-wrapper centers
   343→      // in the full iframe width. This does NOT change the note's page
   344→      // design — only removes the decorative margin the raw file uses
   345→      // when viewed standalone.
   346→      const NEUTRALIZE_ID = "reader-neutralize-body-pad";
   347→      if (!doc.getElementById(NEUTRALIZE_ID)) {
   348→        const s = doc.createElement("style");
   349→        s.id = NEUTRALIZE_ID;
   350→        s.textContent =
   351→          "body { padding: 0 !important; margin: 0 !important; }";
   352→        doc.head.appendChild(s);
   353→      }
   354→
   355→      const avail = container.clientWidth;
   356→      // Small intentional margin so the shadow/rings don't touch the
   357→      // viewport edges. Kept constant in screen px (not scaled).
   358→      const VIEWPORT_MARGIN = 16;
   359→      const targetW = Math.max(0, avail - VIEWPORT_MARGIN * 2);
   360→
   361→      // Find the page element (.page-wrapper / .page / .notebook).
   362→      const pageEl =
   363→        (doc.querySelector(".page-wrapper") as HTMLElement | null) ||
   364→        (doc.querySelector(".page") as HTMLElement | null) ||
   365→        (doc.querySelector(".notebook") as HTMLElement | null) ||
   366→        (doc.body.firstElementChild as HTMLElement | null) ||
   367→        doc.body;
   368→
   369→      // Read the page's CSS width directly (the note uses a fixed
   370→      // width:1080px on .page-wrapper). This is reliable regardless of
   371→      // the iframe's current laid-out width, avoiding chicken-and-egg
   372→      // measurement issues.
   373→      const computedW = doc.defaultView?.getComputedStyle(pageEl).width;
   374→      const parsedW = computedW ? parseFloat(computedW) : 0;
   375→      const pageWNum = isFinite(parsedW) && parsedW > 0 ? parsedW : 1080;
   376→
   377→      // The complete page object width = page + ring overflow (left) +
   378→      // shadow spread (both sides). The rings extend ~40px left of the
   379→      // page; the box-shadow spreads ~60px on all sides.
   380→      const RING_OVERFLOW = 40;
   381→      const SHADOW = 60;
   382→      const objectW = pageWNum + RING_OVERFLOW + SHADOW * 2;
   383→
   384→      // Body height includes the page; add shadow for the bottom.
   385→      const bodyH = Math.max(
   386→        doc.body.scrollHeight,
   387→        doc.body.offsetHeight,
   388→        doc.documentElement.scrollHeight,
   389→        doc.documentElement.offsetHeight,
   390→      );
   391→      const objectH = bodyH + SHADOW;
   392→
   393→      // Base fit-to-width scale (page object fills the viewport).
   394→      const fitScale = targetW < objectW ? targetW / objectW : 1;
   395→      // Apply the user's zoom multiplier. In "fit" mode zoom stays at 1
   396→      // (fitScale alone). In "manual" mode the user's zoom (0.25–4)
   397→      // scales relative to the fit width.
   398→      const zoomMul = zoomMode === "manual" ? zoom : 1;
   399→      const scale = fitScale * zoomMul;
   400→
   401→      // Size the iframe to the complete page object.
   402→      const NATIVE_W = objectW;
   403→      iframe.style.width = `${NATIVE_W}px`;
   404→      iframe.style.height = `${objectH}px`;
   405→      iframe.style.transform = `scale(${scale})`;
   406→      iframe.style.transformOrigin = "top left";
   407→
   408→      // Reserve the scaled footprint on the wrapper so the page can
   409→      // scroll vertically and centers horizontally. When zoomed in
   410→      // beyond the viewport width, the wrapper grows wider than the
   411→      // container and the scroll container handles horizontal scroll.
   412→      const wrapper = iframe.parentElement;
   413→      if (wrapper) {
   414→        wrapper.style.width = `${NATIVE_W * scale}px`;
   415→        wrapper.style.height = `${objectH * scale}px`;
   416→        wrapper.style.marginLeft = "auto";
   417→        wrapper.style.marginRight = "auto";
   418→      }
   419→    } catch {
   420→      /* cross-origin — ignore */
   421→    }
   422→  }, [zoom, zoomMode]);
   423→

   471→  // ─── Page layout (background pattern) ─────────────────────────
   472→  // Inject/replace a style block in the note's document to swap the

   486→          existing.textContent = layout.css;

   488→          const style = doc.createElement("style");
   489→          style.id = STYLE_ID;
   490→          style.textContent = layout.css;
   491→          doc.head.appendChild(style);
   492→        }
   493→      } else if (existing) {
   494→        // "grid" → remove the override so the note's original shows.
   495→        existing.remove();
   496→      }
   497→    } catch {
   498→      /* cross-origin — ignore */
   499→    }
   500→  }, [pageLayout]);
   501→

   503→    applyPageLayout();
   504→  }, [applyPageLayout]);

   515→  React.useEffect(() => {

   518→    return () => document.removeEventListener("fullscreenchange", h);
   519→  }, []);
   520→
   521→  // Close popovers on outside click
   522→  React.useEffect(() => {
   523→    const handler = (e: MouseEvent) => {
   524→      if (downloadOpen && downloadRef.current && !downloadRef.current.contains(e.target as Node)) setDownloadOpen(false);
   525→      if (zoomPresetsOpen) {
